```typescript
export type VagueCheckEstimate = { total: number | null; sections: Array<{ items?: Array<unknown> | null }> | null }
export function isVagueEstimate(e: VagueCheckEstimate | null): boolean {
  const totalNum = e?.total ?? 0
  const hasItems = (e?.sections ?? []).some((s) => (s?.items?.length ?? 0) > 0)
  return totalNum <= 0 || !hasItems
}
```

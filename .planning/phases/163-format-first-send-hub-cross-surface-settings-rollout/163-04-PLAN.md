```typescript
// components/workspace/send/send-hub-dialog.tsx
// Phase 163 (SENDHUB-01, SENDHUB-06): format-first Send hub.
// Three primary format cards (Online Estimate / PDF / Plain Text), each with
// its own delivery actions. Retires the channel-first SendForm tabs AND the
// separate Share & Export menu (see 163-06 for the deletion sweep).
//
// Delivery-action buttons in this file are PLACEHOLDERS -- their onClick shows
// toast.info('Wired in 163-05'). Plan 163-05 replaces them with real server-
// action calls that also record deliveries into estimate_deliveries with the
// widened { format, channel, provider } payload from 163-02's migration.
//
// Mark as Sent IS wired here (SENDHUB-06 requirement of THIS plan).

'use client'

import { useState, useTransition } from 'react'
import { toast } from 'sonner'
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from '@/components/ui/dialog'
import { Button } from '@/components/ui/button'
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card'
import { Copy, ExternalLink, Mail, MessageSquare, Send, Download, FileText } from 'lucide-react'
import { LanguageFlagChip } from './language-flag-chip'
import { PlainTextSheet } from './plain-text-sheet'
import { markAsSentAction } from '@/lib/actions/estimate'
import { buildEstimatePublicPath } from '@/lib/estimate/public-url'
import type { EstimateWithSections } from '@/lib/queries/estimate'
import type { TemplateData } from '@/lib/utils/estimate-template'

export interface SendHubDialogProps {
  open: boolean
  onOpenChange: (v: boolean) => void
  estimate: EstimateWithSections
  projectName: string
  companyName: string
  companySlug?: string | null       // NEW — needed by buildEstimatePublicPath. Wire from estimate-tab in Task 3.
  clientEmail: string | null
  clientPhone: string | null
  clientName: string
  ownerName: string
  estimateTemplate: TemplateData | null
  smsDeliveryEnabled: boolean
  whatsappEnabled?: boolean         // NEW; default false if not passed
}

export function SendHubDialog(props: SendHubDialogProps) {
  const { open, onOpenChange, estimate, projectName, companySlug, ...rest } = props
  const [plainTextOpen, setPlainTextOpen] = useState(false)
  const [isMarkingSent, startTransition] = useTransition()

  const publicPath = buildEstimatePublicPath(
    { slug: companySlug ?? null, name: rest.companyName },
    {
      id: estimate.id,
      public_slug_token: (estimate as { public_slug_token?: string | null }).public_slug_token ?? null,
      share_token: estimate.share_token,
      project_name: projectName,
    }
  )

  function handleMarkAsSent() {
    startTransition(async () => {
      try {
        await markAsSentAction(estimate.id)
        toast.success('Marked as sent')
        onOpenChange(false)
      } catch (err) {
        toast.error(err instanceof Error ? err.message : 'Failed to mark as sent')
      }
    })
  }

  function placeholder(action: string) {
    toast.info(`${action} — wired in 163-05`)
  }

  return (
    <>
      <Dialog open={open} onOpenChange={onOpenChange}>
        <DialogContent className="max-w-3xl">
          <DialogHeader className="flex flex-row items-center justify-between">
            <div>
              <DialogTitle>Send estimate</DialogTitle>
              <DialogDescription>Choose a format and how to deliver it.</DialogDescription>
            </div>
            <LanguageFlagChip lang={estimate.language} />
          </DialogHeader>

          <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
            {/* --- Online Estimate card (default) --- */}
            <Card data-testid="send-hub-card-online-link" className="border-primary/40">
              <CardHeader>
                <CardTitle className="text-base flex items-center gap-2">
                  <ExternalLink className="h-4 w-4" /> Online Estimate
                </CardTitle>
                <CardDescription>Shareable link the client opens in a browser</CardDescription>
              </CardHeader>
              <CardContent className="flex flex-col gap-2">
                <Button variant="outline" size="sm" onClick={() => placeholder('Copy URL')}>
                  <Copy className="h-3.5 w-3.5 mr-2" /> Copy URL
                </Button>
                <Button variant="outline" size="sm" onClick={() => placeholder('Open URL')}>
                  <ExternalLink className="h-3.5 w-3.5 mr-2" /> Open URL
                </Button>
                <Button variant="outline" size="sm" onClick={() => placeholder('Email link')}>
                  <Mail className="h-3.5 w-3.5 mr-2" /> Email
                </Button>
                {rest.smsDeliveryEnabled && (
                  <Button variant="outline" size="sm" onClick={() => placeholder('SMS link')}>
                    <MessageSquare className="h-3.5 w-3.5 mr-2" /> SMS
                  </Button>
                )}
                {rest.whatsappEnabled !== false && (
                  <Button variant="outline" size="sm" onClick={() => placeholder('WhatsApp link')}>
                    <Send className="h-3.5 w-3.5 mr-2" /> WhatsApp
                  </Button>
                )}
              </CardContent>
            </Card>

            {/* --- PDF card --- */}
            <Card data-testid="send-hub-card-pdf">
              <CardHeader>
                <CardTitle className="text-base flex items-center gap-2">
                  <Download className="h-4 w-4" /> PDF
                </CardTitle>
                <CardDescription>Downloadable branded PDF; SMS/WhatsApp fall back to the online link</CardDescription>
              </CardHeader>
              <CardContent className="flex flex-col gap-2">
                <Button variant="outline" size="sm" onClick={() => placeholder('Download PDF')}>
                  <Download className="h-3.5 w-3.5 mr-2" /> Download PDF
                </Button>
                <Button variant="outline" size="sm" onClick={() => placeholder('Email PDF')}>
                  <Mail className="h-3.5 w-3.5 mr-2" /> Email
                </Button>
                {rest.smsDeliveryEnabled && (
                  <Button variant="outline" size="sm" onClick={() => placeholder('SMS PDF (link fallback)')}>
                    <MessageSquare className="h-3.5 w-3.5 mr-2" /> SMS
                  </Button>
                )}
                {rest.whatsappEnabled !== false && (
                  <Button variant="outline" size="sm" onClick={() => placeholder('WhatsApp PDF (link fallback)')}>
                    <Send className="h-3.5 w-3.5 mr-2" /> WhatsApp
                  </Button>
                )}
              </CardContent>
            </Card>

            {/* --- Plain Text card --- */}
            <Card data-testid="send-hub-card-plain-text">
              <CardHeader>
                <CardTitle className="text-base flex items-center gap-2">
                  <FileText className="h-4 w-4" /> Plain Text
                </CardTitle>
                <CardDescription>Plain-text version of the estimate; SMS/WhatsApp fall back to the online link</CardDescription>
              </CardHeader>
              <CardContent className="flex flex-col gap-2">
                <Button variant="outline" size="sm" onClick={() => placeholder('Copy plain text')}>
                  <Copy className="h-3.5 w-3.5 mr-2" /> Copy
                </Button>
                <Button variant="outline" size="sm" onClick={() => setPlainTextOpen(true)}>
                  <FileText className="h-3.5 w-3.5 mr-2" /> Edit
                </Button>
                <Button variant="outline" size="sm" onClick={() => placeholder('Email plain text')}>
                  <Mail className="h-3.5 w-3.5 mr-2" /> Email
                </Button>
                {rest.smsDeliveryEnabled && (
                  <Button variant="outline" size="sm" onClick={() => placeholder('SMS plain text (link fallback)')}>
                    <MessageSquare className="h-3.5 w-3.5 mr-2" /> SMS
                  </Button>
                )}
                {rest.whatsappEnabled !== false && (
                  <Button variant="outline" size="sm" onClick={() => placeholder('WhatsApp plain text (link fallback)')}>
                    <Send className="h-3.5 w-3.5 mr-2" /> WhatsApp
                  </Button>
                )}
              </CardContent>
            </Card>
          </div>

          {/* --- Secondary actions (SENDHUB-06): Mark as Sent + language chip already in header --- */}
          <div className="flex items-center justify-end border-t pt-3 mt-2">
            <Button
              variant="ghost"
              size="sm"
              disabled={isMarkingSent}
              onClick={handleMarkAsSent}
              aria-label="Mark as sent"
            >
              {isMarkingSent ? 'Marking…' : 'Mark as sent'}
            </Button>
          </div>
        </DialogContent>
      </Dialog>

      <PlainTextSheet
        open={plainTextOpen}
        onOpenChange={setPlainTextOpen}
        estimate={estimate}
        clientName={rest.clientName}
        companyName={rest.companyName}
        ownerName={rest.ownerName}
        estimateTemplate={rest.estimateTemplate}
      />
    </>
  )
}
```

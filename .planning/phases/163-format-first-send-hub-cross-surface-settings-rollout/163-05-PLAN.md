```typescript
// Non-throwing -- the caller (hub button) should never see a failure toast for

'use server'
export async function logDeliveryAction(input: {
  estimateId: string
  format: 'online_link' | 'pdf' | 'plain_text'
  channel: 'copy' | 'open' | 'download'
}): Promise<void> {
  try {
    const svc = requireServiceClient()   // or the file's existing service-client factory
    const { data: est, error: estErr } = await svc
      .from('estimates')
      .select('company_id')
      .eq('id', input.estimateId)
      .single()
    if (estErr || !est?.company_id) {
      console.error('[163-05] logDeliveryAction: cannot resolve company_id', estErr)
      return
    }
    await svc.from('estimate_deliveries').insert({
      estimate_id: input.estimateId,
      company_id: est.company_id,
      channel: input.channel,
      format: input.format,
      provider: 'client',
      status: 'sent',
      sent_at: new Date().toISOString(),
    })
  } catch (err) {
    console.error('[163-05] logDeliveryAction failed', err)
  }
}
```

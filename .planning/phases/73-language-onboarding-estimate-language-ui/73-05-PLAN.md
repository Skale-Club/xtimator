**Step 2 — Locale-aware currency and date formatters** (replace or extend the existing static functions):

```typescript
const LOCALE_MAP: Record<EstimateLanguage, string> = {
  en: 'en-US',
  pt: 'pt-BR',
  es: 'es-US',
}

const CURRENCY_MAP: Record<EstimateLanguage, string> = {
  en: 'USD',
  pt: 'BRL',
  es: 'USD',
}

function formatCurrencyLocale(value: number, lang: EstimateLanguage = 'en'): string {
  return new Intl.NumberFormat(LOCALE_MAP[lang], {
    style: 'currency',
    currency: CURRENCY_MAP[lang],
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(value)
}

function formatDateLocale(dateStr: string, lang: EstimateLanguage = 'en'): string {
  return new Date(dateStr).toLocaleDateString(LOCALE_MAP[lang], {
    year: 'numeric',
    month: 'long',
    day: 'numeric',
  })
}
```

Keep the old `formatCurrency` and `formatDate` functions if they are used elsewhere in the file, or replace their usages with the locale-aware versions.

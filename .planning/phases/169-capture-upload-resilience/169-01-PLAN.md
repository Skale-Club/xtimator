// lib/storage/upload-with-retry.ts — sits ON TOP of the StorageProvider abstraction
export async function uploadWithRetry(
  storage: StorageProvider, bucket: string, path: string, blob: Blob,
  opts?: { attempts?: number; baseDelayMs?: number; signal?: AbortSignal }
): Promise<{ path: string }>
// retryable: TypeError/network (no status), 5xx, timeout; NOT retryable: 4xx (auth/policy/size)
// EXCEPTION (Opus check #8): 409 on the SAME path = the object already landed on a prior attempt whose
// response was lost → treat as SUCCESS, not error (runPipeline reuses one storagePath with upsert:false).

// PREREQUISITE (Opus blocker #1): supabase-provider.ts:37-39 currently throws a plain Error, DISCARDING
// error.status/statusCode from @supabase/storage-js. Minimal provider change (allowed — scope fence updated):
//   throw Object.assign(new Error(msg), { status: error.status, statusCode: error.statusCode, cause: error })
// Message/throw contract unchanged; the wrapper classifies on the preserved .status.

// lib/capture/blob-store.ts — hand-rolled IDB (no dependency), ALL methods never-throw (resolve null/false on failure)
// Opus check #4: store an ArrayBuffer, NOT a raw Blob — iOS Safari (the target platform) has a history of
// Blobs in IDB coming back unreadable/zero-length. Reconstruct new Blob([buffer], {type: mimeType}) on read.
export interface PendingCapture {
  key: string            // `${projectKey}` — one pending capture per project/flow
  buffer: ArrayBuffer
  mimeType: string
  durationSeconds: number
  createdAt: number      // Date.now() at save
}
export async function savePendingCapture(p: PendingCapture): Promise<boolean>
export async function getPendingCapture(key: string): Promise<PendingCapture | null>
export async function deletePendingCapture(key: string): Promise<void>

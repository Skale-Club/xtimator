```ts
// lib/ai/photo-extraction-schema.ts
export const photoExtractionSchema = z.object({
  version: z.literal(1).catch(1),
  surfaces: z.array(z.object({ name: z.string(), material: z.string().nullable().catch(null), condition: z.string().nullable().catch(null) })).catch([]),
  measurements: z.array(z.object({
    dimension: z.enum(['length','area','height','count']),
    value: z.number().finite().positive(),
    unit: z.string().min(1),
    subject: z.string().min(1),
    confidence: z.enum(['stated','estimated']).catch('estimated'),
  })).catch([]),
  materials: z.array(z.string()).catch([]),
  damage: z.array(z.object({ description: z.string().min(1), severity: z.enum(['minor','moderate','severe']).catch('moderate') })).catch([]),
  trade_signals: z.array(z.string()).catch([]),
  access_notes: z.string().optional().nullable(),
  overall_description: z.string().min(1),   // THE hard requirement — populates ai_description
})
export type PhotoExtraction = z.infer<typeof photoExtractionSchema>
// JSON-schema mirror for the provider tool declarations (advisory layer) — exported so
// 171-02's OpenRouter tool + Gemini functionDeclaration derive from ONE source and the
// PEXT-04 parity test can assert against it. Hand-written object (house style — the
// estimate schema does the same), NOT a zod-to-jsonschema dependency.
export function photoExtractionToolSchema(): Record<string, unknown>
```
Defensive posture: element-level `.catch()` on enums/optionals so one malformed measurement doesn't reject the whole extraction; but an invalid measurements ARRAY (e.g. value non-numeric after coercion) drops the bad element rather than the parse — implement via array-level preprocess filtering elements that fail their element schema, mirroring how the estimate schema's price_source preprocess never rejects the whole parse. overall_description missing/empty → parse FAILS (the caller falls back to prose — correct).
Migration: nullable JSONB only, comment documenting the dormant-first contract + schema version key. NO backfill.

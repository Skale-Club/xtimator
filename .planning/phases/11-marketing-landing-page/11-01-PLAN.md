import { describe, it, expect } from 'vitest'
import { render, screen } from '@testing-library/react'
import { HeroSection } from '@/components/landing/hero-section'
import { HowItWorksSection } from '@/components/landing/how-it-works-section'
import { FeaturesSection } from '@/components/landing/features-section'

describe('LAND-01: Hero section', () => {
  it('renders the headline', () => {
    render(<HeroSection />)
    expect(screen.getByRole('heading', { level: 1 })).toHaveTextContent(
      'Professional estimates in 5 minutes.'
    )
  })

  it('renders the subheadline text', () => {
    render(<HeroSection />)
    expect(screen.getByText(/Record a job site walkthrough/)).toBeTruthy()
  })

  it('renders Get Started Free CTA linking to /auth/signup', () => {
    render(<HeroSection />)
    const cta = screen.getByRole('link', { name: /Get Started Free/i })
    expect(cta).toBeTruthy()
    expect(cta.getAttribute('href')).toBe('/auth/signup')
  })

  it('renders Sign In secondary link to /auth/login', () => {
    render(<HeroSection />)
    const signIn = screen.getByRole('link', { name: /Sign In/i })
    expect(signIn.getAttribute('href')).toBe('/auth/login')
  })
})

describe('LAND-02: How It Works section', () => {
  it('renders the How It Works heading', () => {
    render(<HowItWorksSection />)
    expect(screen.getByRole('heading', { name: /How It Works/i })).toBeTruthy()
  })

  it('renders 3 step card titles', () => {
    render(<HowItWorksSection />)

    expect(screen.getByText('Add photos')).toBeTruthy()

})

describe('LAND-03: Features section', () => {

  })
})

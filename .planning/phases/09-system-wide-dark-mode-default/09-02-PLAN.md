      test.beforeEach(async ({ context }) => {
        // Ensure no pre-existing eb-theme cookie
        await context.clearCookies()
      })

      test('default-dark: fresh visit renders <html class="dark">', async ({ page }) => {
        // /auth/login is public and uses the root layout's ThemeProvider.
        // With no cookie, next-themes honors defaultTheme="dark".
        await page.goto('/auth/login')
        await expect.poll(async () =>

      test('scoped-wrappers-intact: /auth/login still has [data-theme="dark-auth"]', async ({ page }) => {

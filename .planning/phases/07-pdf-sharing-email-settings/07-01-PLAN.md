    ```typescript
    export async function getEstimateWithContext(supabase: SupabaseClient, estimateId: string) {
      // Fetch estimate with sections+items
      const estimate = await getEstimateById(supabase, estimateId)
      if (!estimate) return null

      // Fetch project with client join
      const { data: project } = await supabase
        .from('projects')
        .select('name, project_type, client:clients(name, email, phone, address, city, state, zip)')
        .eq('id', estimate.project_id)
        .single()

      // Fetch company
      const { data: company } = await supabase
        .from('companies')
        .select('name, owner_name, phone, email, website, address, city, state, zip, logo_url, brand_primary_color')
        .eq('id', estimate.company_id)
        .single()

      return { estimate, project, company }
    }
    ```

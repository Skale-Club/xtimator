<objective>
Ship the `/admin/branding` page alongside Plans 04 and 06 as a parallel Wave-3 sibling. Admin can edit app name, upload logo, change primary color, and preview the changes live before saving. On save, the singleton `platform_branding.id=1` row is upserted, the logo lands in the `platform-brand` Storage bucket, and `invalidatePlatformConfig()` clears the loader cache so downstream pages pick up the change within one request.

    1. Create `app/admin/branding/actions.ts`:
       ```ts
       'use server'
       import { z } from 'zod'
       import { revalidatePath } from 'next/cache'
       import { requireAdmin } from '@/lib/auth/admin-context'
       import { createServiceClient } from '@/lib/supabase/service'
       import { invalidatePlatformConfig } from '@/lib/platform-config'
       import { brandingSchema } from '@/lib/schemas/admin'

       export async function saveBranding(formData: FormData) {
         const ctx = await requireAdmin()
         const raw = {
           appName: formData.get('appName'),
           primaryColor: formData.get('primaryColor') || null,
           emailFromName: formData.get('emailFromName') || null,
           logoFile: formData.get('logoFile') as File | null,
         }
         const parsed = brandingSchema.safeParse(raw)
         if (!parsed.success) return { ok: false, errors: parsed.error.flatten() }

         const svc = createServiceClient()
         let logoUrl: string | undefined = undefined
         if (parsed.data.logoFile && parsed.data.logoFile.size > 0) {
           const f = parsed.data.logoFile
           const ext = (f.name.split('.').pop() || 'png').toLowerCase()
           const path = `logo-${Date.now()}.${ext}`
           const body = Buffer.from(await f.arrayBuffer())
           const { data: up, error: upErr } = await svc.storage
             .from('platform-brand')
             .upload(path, body, { contentType: f.type, upsert: true })
           if (upErr) return { ok: false, message: upErr.message }
           const { data: pub } = svc.storage.from('platform-brand').getPublicUrl(up!.path)
           logoUrl = pub.publicUrl
         }

         const { error: dbErr } = await svc
           .from('platform_branding')
           .upsert({
             id: 1,
             app_name: parsed.data.appName,
             ...(logoUrl !== undefined ? { logo_url: logoUrl } : {}),
             primary_color: parsed.data.primaryColor,
             email_from_name: parsed.data.emailFromName,
             updated_at: new Date().toISOString(),
             updated_by: ctx.userId,
           })
         if (dbErr) return { ok: false, message: dbErr.message }

         invalidatePlatformConfig()
         revalidatePath('/admin/branding')
         revalidatePath('/', 'layout')
         return { ok: true }
       }
       ```

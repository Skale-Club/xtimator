       import { requireAdmin } from '@/lib/auth/admin-context'
       import { createServiceClient } from '@/lib/supabase/service'
       import { decrypt } from '@/lib/crypto/aes'
       import { IntegrationCard } from './integration-card'

       const PROVIDERS = [
         { id: 'resend' as const, title: 'Resend', description: 'Transactional email delivery (estimate sends, notifications).' },
         { id: 'anthropic' as const, title: 'Anthropic', description: 'Claude models for estimate generation and photo analysis.' },
         { id: 'openai' as const, title: 'OpenAI', description: 'Whisper API for audio transcription.' },
       ]

       export default async function IntegrationsPage() {
         await requireAdmin()
         const svc = createServiceClient()
         const { data: rows } = await svc
           .from('platform_integrations')
           .select('provider, ciphertext, iv, auth_tag, updated_at, updated_by')
         const byProvider = new Map<string, {configured:true, last4:string, updatedAt:string, updatedByEmail:string}>()
         for (const r of rows ?? []) {
           try {
             const pt = decrypt({ ciphertext: Buffer.from(r.ciphertext), iv: Buffer.from(r.iv), authTag: Buffer.from(r.auth_tag) })
             const { data: u } = r.updated_by
               ? await svc.auth.admin.getUserById(r.updated_by)
               : { data: null }
             byProvider.set(r.provider, {
               configured: true,
               last4: pt.slice(-4),
               updatedAt: r.updated_at,
               updatedByEmail: u?.user?.email ?? '',
             })
           } catch { /* decrypt failed — treat as unconfigured */ }
         }
         return (
           <>
             <h1 className="text-xl font-semibold">Integrations</h1>
             <p className="text-muted-foreground mt-1">Manage API keys for services shared across all tenants. Changes take effect within 60 seconds.</p>
             <div className="mt-8 flex flex-col gap-6">
               {PROVIDERS.map(p => (
                 <IntegrationCard
                   key={p.id}
                   provider={p.id}
                   title={p.title}
                   description={p.description}
                   initial={byProvider.get(p.id) ?? { configured: false }}
                 />
               ))}
             </div>
           </>
         )
       }

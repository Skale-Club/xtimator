```typescript
// lib/notifications/template-engine.ts (NEW)

export type RenderChannel = 'html' | 'text'
export type TemplateVars = Record<string, string | number | undefined | null>

/** HTML-entity-escape one value: &, <, >, ", ' — mirrors lib/email/notification-emails.ts's escapeHtml(). */
export function escapeHtmlValue(value: string): string

/** Plain-text sanitize one value for SMS/in-app: strip control chars (incl. \r\n\t), collapse
 *  runs of 2+ spaces to one, trim. NO HTML-entity escaping (SMS must never show literal '&amp;'). */
export function escapeTextValue(value: string): string

/** Sanitize ONE WhatsApp HSM {{n}} positional parameter: Meta rejects/mishandles newlines, tabs,
 *  4+ consecutive spaces, and leading/trailing whitespace (research/STACK.md, Pitfall 3 context). */
export function sanitizeWhatsAppParam(value: string): string

/** Extract the ordered UNIQUE list of {{var}} names referenced in a template string
 *  (feeds the future admin-editor variable catalog, TMPL-03 — build now, no consumer yet). */
export function extractVariables(template: string): string[]

/**
 * Substitute every {{var}} token in `template` with `vars[var]`, escaping the VALUE
 * (never the template text) per `channel`. Missing/undefined/null -> ''. Never throws —
 * a malformed/unmatched '{{' with no closing '}}' just fails the regex and passes
 * through as literal text (no code execution risk, just literal output).
 */
export function renderTemplate(template: string, vars: TemplateVars, channel: RenderChannel): string
```

Regex is `/\{\{(\w+)\}\}/g` (word chars only inside the braces — this is WHY a nested/malformed case like `'{{{{name}}}}'` resolves to `'{{' + rendered(name) + '}}'`: the innermost complete `{{name}}` is the only thing `\w+` can match; leftover outer braces pass through literally). Lock this exact behavior with a test — it is the plan's one genuinely non-obvious edge case.

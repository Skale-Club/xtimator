```typescript
// lib/notifications/template-resolver.ts (NEW)
import type { EventType } from './event-types'
import { buildNotificationCopy, type CopyContext, type NotificationCopy } from './copy'
import { renderTemplate, type TemplateVars } from './template-engine'

export type TemplateScope = 'tenant' | 'customer'
export type TemplateChannel = 'in_app' | 'email' | 'sms'

/**
 * DB-first, copy.ts-fallback resolver. Mirrors whatsapp-registry.ts's
 * getApprovedTemplateForEvent(): lazy-import createServiceClient (keeps this
 * module server-client-free for static analysis, same reasoning as the
 * WhatsApp registry), no client -> fallback, no/inactive row -> fallback,
 * empty-after-render body -> fallback (never send blank content, Pitfall 2),
 * ANY throw -> caught, warned, fallback. NEVER throws out of this function.
 */
export async function resolveNotificationCopy(
  scope: TemplateScope,
  eventType: EventType,
  channel: TemplateChannel,
  ctx: CopyContext,
): Promise<NotificationCopy>
```

Query shape (mirrors `getApprovedTemplateForEvent`'s `.select().eq().eq().eq().eq().maybeSingle()` chain from `whatsapp-registry.ts`):
```typescript
const svc = createServiceClient()
if (!svc) return buildNotificationCopy(eventType, ctx)
const { data } = await svc
  .from('notification_templates')
  .select('title, subject, body')
  .eq('scope', scope)
  .eq('event_type', eventType)
  .eq('channel', channel)
  .eq('is_active', true)
  .maybeSingle()
// row.body null/missing -> fallback. Otherwise render with mode = channel==='email' ? 'html' : 'text'.
// label = channel === 'email' ? row.subject : row.title  (subject is email-only, title is in_app-only per the 172-02 schema)
// final body = renderTemplate(row.body, ctx as TemplateVars, mode).trim() — if '', fallback (Pitfall 2).
```

```typescript
let resolvedTitle = params.title
let resolvedBody = params.body
if (params.copyContext) {
  try {
    const copy = await resolveNotificationCopy('tenant', params.eventType, 'in_app', params.copyContext)
    resolvedTitle = copy.title
    resolvedBody = copy.body
  } catch (e) {
    console.warn('[notifications.dispatch] copyContext resolution failed, using caller-supplied title/body:', e instanceof Error ? e.message : String(e))
    // resolvedTitle/resolvedBody already default to params.title/params.body — no-op fallback.
  }
}
```
Then replace every downstream use of `params.title`/`params.body` (the in_app insert, the email Inngest payload's `title`/`body`, the sms `` `${params.title}: ${params.body}` `` format string, and the WhatsApp branch's `tpl.variables({ title: params.title, body: params.body })`) with `resolvedTitle`/`resolvedBody`. Because `resolveNotificationCopy` is ALREADY internally never-throwing (its own try/catch falls back to `buildNotificationCopy`), the outer try/catch here is defense-in-depth ONLY (mirrors the existing double-guard style already used for the WhatsApp branch's `getApprovedTemplateForEvent` call, `dispatch.ts:186-211`) — it should never actually trigger in practice, but keep it for the same "Research Pitfall 4" reasoning already documented in the file.

export interface ClientMatch { id: string; name: string | null; phone: string | null; email: string | null }

  companyId: string

  channel: 'email' | 'sms'
  body: string                  // LLM-authored or owner-dictated, untrusted

  channelRef?: string | null

export type DraftSendResult =
  | { ok: true; confirmationId: string; token: string | null; clientId: string; clientName: string | null
      recipient: string; channel: 'email' | 'sms'; body: string; subject?: string }
  | { ok: false; error: 'client_not_found' | 'client_ambiguous' | 'no_recipient_email' | 'no_recipient_phone' | 'rate_limited'
      candidates?: string[] }

export async function draftCustomerMessage(supabase: SupabaseClient, params: DraftSendParams): Promise<DraftSendResult>

export interface ConfirmSendResult { ok: boolean; error?: string }
export async function confirmSendByChannelRef(supabase: SupabaseClient, companyId: string, channelRef: string): Promise<ConfirmSendResult>
export async function cancelSendByChannelRef(supabase: SupabaseClient, companyId: string, channelRef: string): Promise<{ ok: boolean }>
export async function confirmSendByToken(supabase: SupabaseClient, companyId: string, token: string): Promise<ConfirmSendResult>

Internal flow (LOAD-BEARING order) for `draftCustomerMessage`:
1. `checkAgenticSendRateLimit(params.companyId)` -> `false` -> `{ ok: false, error: 'rate_limited' }`. No client lookup, no DB write.
2. `clients` lookup: `.select('id,name,phone,email').eq('company_id', params.companyId).ilike('name', \`%${clientQuery}%\`).limit(5)`. 0 rows -> `client_not_found`. 2+ rows -> `client_ambiguous` with `candidates: names`. Exactly 1 -> proceed.
3. `channel==='email'` and no `client.email` -> `no_recipient_email`. `channel==='sms'` and no `client.phone` -> `no_recipient_phone`.
4. `createSendConfirmation(supabase, { companyId, clientId: client.id, channel, subject, body, triggerSource, channelRef })`.
5. Return the resolved preview (client name + recipient + exact body/subject + confirmationId + token).

Internal flow for `finalizeConfirmedSend(supabase, pending)` (module-private, shared by both public confirm-by-* functions):
1. `verifyBinding(pending)` false -> `markRefused` -> `{ ok: false, error: 'integrity_error' }`.
2. `assertSendAllowed(pending.companyId, pending.clientId, pending.channel)` -> not allowed -> `markRefused` -> `{ ok: false, error: gate.reason }`.
3. Resolve `businessName` from `companies.name` (fallback `'Your contractor'` if missing).
4. `sendCustomerMessage({ permit, companyId, clientId, businessName, triggerSource: pending.triggerSource, freeform: { subject: pending.subject ?? undefined, body: pending.body } })`.
5. `markConfirmed(supabase, pending.id)` — runs regardless of the dispatch result (a provider-level failure is still an "owner confirmed and we attempted it" event, audited via `customer_messages` by step 4's own logging).
6. Return `{ ok: result.ok, error: result.error }`.

`confirmSendByChannelRef`/`confirmSendByToken` both: resolve the pending row (`resolvePendingByChannelRef`/`resolveByToken`) -> `null` -> `{ ok: false, error: 'not_found' }` -> else `finalizeConfirmedSend(supabase, pending)`.
`cancelSendByChannelRef`: resolve the pending row -> `null` -> `{ ok: false }` -> else `markCancelled(supabase, pending.id)` -> `{ ok: true }`.

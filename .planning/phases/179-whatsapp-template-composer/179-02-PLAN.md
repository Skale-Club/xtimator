function mapMetaEventToStatus(event: string): string {
  switch (event.toUpperCase()) {
    case 'APPROVED': return 'approved'
    case 'REJECTED': return 'rejected'
    case 'PENDING':
    case 'PENDING_DELETION': return 'pending'

}

    ```typescript
    import { redirect } from 'next/navigation'
    import { Eye } from 'lucide-react'
    import { startSupportSession } from '@/lib/auth/support-mode'

    interface SupportModeButtonProps {
      companyId: string
    }

    async function startSupportSessionAction(formData: FormData) {
      'use server'
      const companyId = formData.get('companyId')
      if (typeof companyId !== 'string' || !companyId) return
      await startSupportSession(companyId)
      redirect('/dashboard')
    }

    /**
     * Companies-list row action — starts a Support Mode session for this company
     * and redirects the admin into the tenant's /dashboard view (SUPPORT-01).
     * Visually distinct from HandoffButton (Send icon, "Hand off") and the plain
     * "Configure →" text link — per 151-UI-SPEC.md.
     */
    export function SupportModeButton({ companyId }: SupportModeButtonProps) {
      return (
        <form action={startSupportSessionAction}>
          <input type="hidden" name="companyId" value={companyId} />
          <button
            type="submit"
            className="inline-flex items-center gap-1 text-xs text-[hsl(var(--primary))] hover:underline font-medium"
          >
            <Eye className="h-3 w-3" />
            Support Mode →
          </button>
        </form>
      )
    }
    ```

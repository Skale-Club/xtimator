export const projectSchema = z.object({
  clientId: z.string().min(1, 'Please select a client'),
  clientName: z.string(),
  name: z.string().min(1, 'Project name is required').max(100, 'Project name too long'),
  projectType: z.string().min(1, 'Please select a project type'),
  customProjectType: z.string().optional().or(z.literal('')),
  targetBudget: z.string().optional().or(z.literal('')),
})

export const STEP_FIELDS: Record<number, (keyof ProjectFormValues)[]> = {

}

- Single `useForm<ProjectFormValues>` with `zodResolver(projectSchema) as any` (Pitfall 1), mode `'onBlur'`

- State: `const [currentStep, setCurrentStep] = useState(1)`

    - Step 2: `<StepProjectDetails form={form} projectTypes={projectTypes} />`
    - Step 3: `<StepConfirmation form={form} />`
  - `Separator`
  - Navigation footer: Back button (step > 1), Next button (step < 3), "Create Project" button (step 3, with Loader2 spinner when isPending)
  - All buttons: `min-h-[44px]` (UX-02)

**Create `components/projects/step-project-details.tsx`** ('use client', per D-03):
- Props: `{ form: UseFormReturn<ProjectFormValues>; projectTypes: string[] }`
- Section heading: "Project Details"
- **Project name** field (FormField + Input):
  - Auto-suggestion logic (per D-03 and Pitfall 3):
    - Track `nameManuallyEdited` with `useRef<boolean>(false)`
    - Track `lastAutoSuggestion` with `useRef<string>('')`
    - Watch `clientName` and `projectType` with `form.watch`
    - `useEffect` on `[clientName, projectType]`: if `!nameManuallyEdited.current`, compute suggestion = `${clientName} - ${effectiveType}` (where effectiveType = customProjectType if type is Custom, else projectType). Set `form.setValue('name', suggestion)` and update `lastAutoSuggestion.current`
    - On name field `onChange`: if new value differs from `lastAutoSuggestion.current`, set `nameManuallyEdited.current = true`. If value matches current auto-suggestion, keep `false`.
    - When client or type changes and name matches `lastAutoSuggestion.current`, reset `nameManuallyEdited.current = false` so new auto-suggestion applies
- **Project type** dropdown (FormField + Select from shadcn/ui):
  - `SelectTrigger` with `SelectValue` placeholder "Select project type"
  - Map `projectTypes` to `SelectItem` entries
  - Add final `SelectItem` with value "Custom" and label "Custom..."
  - On change: `form.setValue('projectType', value)`. If value !== 'Custom', clear customProjectType
- **Custom project type** (conditional, per PROJ-05):
  - Show `Input` field only when `form.watch('projectType') === 'Custom'`
  - Placeholder: "Enter custom project type"
  - Bind to `customProjectType` form field
- **Target budget** field (FormField + Input, per PROJ-06):
  - Type "text", inputMode "decimal"
  - Placeholder: "Optional (e.g., 5000)"
  - Prefix "$" label or adornment
  - Bind to `targetBudget` form field

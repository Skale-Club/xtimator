    export function detectColumnMapping(headers: string[]): Record<string, TargetField | '_skip'> {
      const result: Record<string, TargetField | '_skip'> = {}
      const claimedTargets = new Set<TargetField>()

      // Priority order: name > unit_price > folder > unit > notes
      // Each target claims the first matching header; later matches with same target → '_skip'
      const priority: TargetField[] = ['name', 'unit_price', 'folder', 'unit', 'notes']
      const normalize = (h: string) => h.trim().toLowerCase()

      for (const target of priority) {
        const aliases = ALIAS_DICTIONARY[target]
        for (const rawHeader of headers) {
          const norm = normalize(rawHeader)
          if (aliases.includes(norm) && !claimedTargets.has(target) && !(rawHeader in result)) {
            result[rawHeader] = target
            claimedTargets.add(target)
            break
          }
        }
      }
      // Any unmapped header → '_skip'
      for (const h of headers) {
        if (!(h in result)) result[h] = '_skip'
      }
      return result
    }

    const SYMBOL_RE = /[$€£¥]|R\$/g

    function stripSymbolsAndQuotes(raw: string): string {
      return raw.trim().replace(/^["']|["']$/g, '').replace(SYMBOL_RE, '').trim()
    }

    export function detectLocale(samples: string[]): LocaleMode {
      let usScore = 0, brScore = 0, plainScore = 0
      for (const raw of samples.slice(0, 5)) {
        const s = stripSymbolsAndQuotes(raw)
        if (/^\d{1,3}(,\d{3})+(\.\d+)?$/.test(s)) usScore++
        else if (/^\d{1,3}(\.\d{3})+(,\d+)?$/.test(s)) brScore++
        else if (/^\d+(\.\d+)?$/.test(s)) plainScore++
        else if (/^\d+,\d+$/.test(s)) brScore++ // "99,00"
      }
      if (brScore > usScore && brScore >= plainScore) return 'br'
      if (usScore >= brScore && usScore > 0) return 'us'
      return 'plain'
    }

    export function parseCurrency(raw: string, mode: LocaleMode, custom?: CustomLocale): number | null {
      if (!raw || !raw.trim()) return null
      let s = stripSymbolsAndQuotes(raw).replace(/\s/g, '')
      if (!s) return null

      switch (mode) {
        case 'us':
          s = s.replace(/,/g, '')
          break
        case 'br':
          s = s.replace(/\./g, '').replace(/,/g, '.')
          break
        case 'plain':
          // leave as-is
          break
        case 'custom':
          if (!custom) return null
          // strip thousands, then swap decimal for '.'
          if (custom.thousands) s = s.split(custom.thousands).join('')
          if (custom.decimal !== '.') s = s.replace(custom.decimal, '.')
          break
      }
      const n = Number(s)
      if (Number.isNaN(n)) return null
      return Math.round(n * 100) / 100
    }

    import type { ParsedRow } from './price-book-import'

    export type DedupeStrategy = 'skip' | 'update' | 'new'

    export interface DedupeInput {
      rows: ParsedRow[]
      existingKeys: Set<string>
      existingNamesByFolder: Map<string, Set<string>>
      globalStrategy: DedupeStrategy
      perRowOverrides: Map<number, DedupeStrategy>
    }

    export type DedupeAction =
      | { rowNumber: number; action: 'insert'; finalName: string }
      | { rowNumber: number; action: 'update'; finalName: string }
      | { rowNumber: number; action: 'skip'; reason: 'dedupe' | 'invalid' | 'in_file_dup' }

    function suffixUntilFree(name: string, taken: Set<string>): string {
      let n = 2
      let candidate = `${name} (${n})`
      while (taken.has(candidate.toLowerCase())) {
        n++
        candidate = `${name} (${n})`
      }
      return candidate
    }

    export function applyDedupeStrategy(input: DedupeInput): DedupeAction[] {
      const out: DedupeAction[] = []
      // Mutate a copy of existingNamesByFolder so 'new' suffixes stack correctly within batch
      const namesByFolder = new Map<string, Set<string>>()
      for (const [k, v] of input.existingNamesByFolder) namesByFolder.set(k, new Set(v))

      for (const row of input.rows) {
        if (row.errors.length > 0) {
          out.push({ rowNumber: row.rowNumber, action: 'skip', reason: 'invalid' })
          continue
        }
        if (row.isDuplicateInFile) {
          out.push({ rowNumber: row.rowNumber, action: 'skip', reason: 'in_file_dup' })
          continue
        }
        const folderKey = (row.values.folder_name ?? '').toLowerCase()
        const nameLower = row.values.name.toLowerCase()
        const key = `${folderKey}::${nameLower}`
        const collides = input.existingKeys.has(key)

        if (!collides) {
          out.push({ rowNumber: row.rowNumber, action: 'insert', finalName: row.values.name })
          continue
        }

        const strategy = input.perRowOverrides.get(row.rowNumber) ?? input.globalStrategy
        if (strategy === 'skip') {
          out.push({ rowNumber: row.rowNumber, action: 'skip', reason: 'dedupe' })
        } else if (strategy === 'update') {
          out.push({ rowNumber: row.rowNumber, action: 'update', finalName: row.values.name })
        } else {
          // 'new' — suffix
          const taken = namesByFolder.get(folderKey) ?? new Set<string>()
          const finalName = suffixUntilFree(row.values.name, taken)
          taken.add(finalName.toLowerCase())
          namesByFolder.set(folderKey, taken)
          out.push({ rowNumber: row.rowNumber, action: 'insert', finalName })
        }
      }
      return out
    }
    ```

    import type { ParsedRow, ImportRow } from './price-book-import'
    import type { LocaleMode, CustomLocale } from './locale-parser'
    import type { TargetField } from './price-book-aliases'
    import { detectColumnMapping } from './price-book-aliases'

    export type WizardStep = 'upload' | 'map' | 'preview' | 'confirm'
    export type DedupeStrategy = 'skip' | 'update' | 'new'

    export interface WizardState {
      step: WizardStep
      fileName: string | null
      headers: string[]
      rows: ParsedRow[]
      locale: LocaleMode
      customLocale?: CustomLocale
      mapping: Record<string, TargetField | '_skip'>
      globalDedupeStrategy: DedupeStrategy
      perRowDedupeOverrides: Record<number, DedupeStrategy>
      cellEdits: Record<number, Partial<ImportRow>>
      savedAt: number
    }

    export const initialWizardState: WizardState = {
      step: 'upload',
      fileName: null,
      headers: [],
      rows: [],
      locale: 'plain',
      mapping: {},
      globalDedupeStrategy: 'skip',
      perRowDedupeOverrides: {},
      cellEdits: {},
      savedAt: 0,
    }

    export type WizardAction =
      | { type: 'FILE_PARSED'; fileName: string; headers: string[]; rows: ParsedRow[]; detectedLocale: LocaleMode }
      | { type: 'LOCALE_SET'; mode: LocaleMode; custom?: CustomLocale }
      | { type: 'MAPPING_SET'; header: string; target: TargetField | '_skip' }
      | { type: 'MAPPING_COMPLETE' }
      | { type: 'EDIT_CELL'; rowNumber: number; field: keyof ImportRow; value: string | number }
      | { type: 'DEDUPE_STRATEGY_SET'; strategy: DedupeStrategy }
      | { type: 'DEDUPE_ROW_OVERRIDE'; rowNumber: number; strategy: DedupeStrategy }
      | { type: 'PREVIEW_COMPLETE' }
      | { type: 'BACK' }
      | { type: 'RESET' }
      | { type: 'RESTORE_DRAFT'; state: WizardState }

    export function wizardReducer(state: WizardState, action: WizardAction): WizardState {
      const stamp = (s: WizardState): WizardState => ({ ...s, savedAt: Date.now() })
      switch (action.type) {
        case 'FILE_PARSED':
          return stamp({
            ...state,
            step: 'map',
            fileName: action.fileName,
            headers: action.headers,
            rows: action.rows,
            locale: action.detectedLocale,
            mapping: detectColumnMapping(action.headers),
          })
        case 'LOCALE_SET':
          return stamp({ ...state, locale: action.mode, customLocale: action.custom })
        case 'MAPPING_SET':
          return stamp({ ...state, mapping: { ...state.mapping, [action.header]: action.target } })
        case 'MAPPING_COMPLETE':
          return stamp({ ...state, step: 'preview' })
        case 'EDIT_CELL': {
          const existing = state.cellEdits[action.rowNumber] ?? {}
          return stamp({
            ...state,
            cellEdits: {
              ...state.cellEdits,
              [action.rowNumber]: { ...existing, [action.field]: action.value },
            },
          })
        }
        case 'DEDUPE_STRATEGY_SET':
          return stamp({ ...state, globalDedupeStrategy: action.strategy })
        case 'DEDUPE_ROW_OVERRIDE':
          return stamp({
            ...state,
            perRowDedupeOverrides: { ...state.perRowDedupeOverrides, [action.rowNumber]: action.strategy },
          })
        case 'PREVIEW_COMPLETE':
          return stamp({ ...state, step: 'confirm' })
        case 'BACK': {
          const flow: WizardStep[] = ['upload', 'map', 'preview', 'confirm']
          const idx = flow.indexOf(state.step)
          return stamp({ ...state, step: flow[Math.max(0, idx - 1)] })
        }
        case 'RESET':
          return { ...initialWizardState }
        case 'RESTORE_DRAFT':
          return action.state
        default:
          return state
      }
    }

    const DRAFT_VERSION = 1
    const DRAFT_TTL_MS = 24 * 60 * 60 * 1000  // 24h

    export function serializeDraft(state: WizardState): string {
      return JSON.stringify({ version: DRAFT_VERSION, state })
    }

    export function deserializeDraft(json: string): WizardState | null {
      try {
        const parsed = JSON.parse(json) as { version: number; state: WizardState }
        if (parsed.version !== DRAFT_VERSION) return null
        if (Date.now() - parsed.state.savedAt > DRAFT_TTL_MS) return null
        return parsed.state
      } catch {
        return null
      }
    }
    ```

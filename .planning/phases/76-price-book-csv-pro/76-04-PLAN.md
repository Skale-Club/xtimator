    ```typescript
    import type { ImportRow } from './price-book-import'

    export interface FailedRow {
      values: ImportRow
      reason: string
    }

    function escapeCell(v: unknown): string {
      const s = v === null || v === undefined ? '' : String(v)
      if (/[",\n\r]/.test(s)) return `"${s.replace(/"/g, '""')}"`
      return s
    }

    export function buildErrorCsv(rows: FailedRow[], headers: string[]): string {
      const fullHeaders = [...headers, 'error_reason']
      const headerLine = fullHeaders.map(escapeCell).join(',')
      const dataLines = rows.map((r) => {
        const cells = headers.map((h) => {
          const v = (r.values as Record<string, unknown>)[h] ?? (r.values as Record<string, unknown>)[h === 'folder' ? 'folder_name' : h]
          return escapeCell(v)
        })
        cells.push(escapeCell(r.reason))
        return cells.join(',')
      })
      return [headerLine, ...dataLines].join('\n')
    }
    ```

    ```typescript
    import { applyDedupeStrategy, type DedupeStrategy, type DedupeAction } from '@/lib/csv/dedupe'
    import type { ImportRow, ParsedRow } from '@/lib/csv/price-book-import'

    export interface CommitChunkInput {
      rows: ImportRow[]
      globalStrategy: DedupeStrategy
      perRowOverrides: Record<number, DedupeStrategy>
      folderNameMap?: Record<string, string>
      chunkIndex: number
      totalChunks: number
      importId?: string
      filename: string
      locale: string
      dryRun?: boolean
    }

    export interface CommitChunkResult {
      importId: string
      chunkIndex: number
      totalChunks: number
      insertedCount: number
      updatedCount: number
      skippedCount: number
      failedRows: { rowNumber: number; values: ImportRow; reason: string }[]
      done: boolean
    }

    export async function commitImportChunk(
      input: CommitChunkInput
    ): Promise<{ data: CommitChunkResult } | { error: string }> {
      const ctx = await getAuthContext()
      if ('error' in ctx) return { error: ctx.error }
      const { supabase, company } = ctx

      // 1. Resolve folders (insert any missing) — folder names map already may include some
      const folderNames = [...new Set(input.rows.map((r) => r.folder_name).filter(Boolean) as string[])]
      let folderMap: Map<string, string> = new Map(
        Object.entries(input.folderNameMap ?? {})
      )
      const insertedFolderIds: string[] = []
      if (folderNames.length > 0) {
        const missing = folderNames.filter((n) => !folderMap.has(n.toLowerCase()))
        if (missing.length > 0 && !input.dryRun) {
          const folderResult = await resolveOrCreateFolders(missing)
          if ('error' in folderResult) return { error: folderResult.error }
          for (const [k, v] of folderResult.data) {
            if (!folderMap.has(k)) {
              folderMap.set(k, v)
              insertedFolderIds.push(v)
            }
          }
        }
      }

      // 2. Fetch existing keys + names per folder for dedupe
      const { data: existing, error: exErr } = await supabase
        .from('company_price_book')
        .select('id, folder_id, name, unit, unit_price, notes')
        .eq('company_id', company.id)
      if (exErr) return { error: 'Could not load existing items.' }

      const folderIdToName = new Map<string | null, string>()
      for (const [name, id] of folderMap) folderIdToName.set(id, name)

      const existingByKey = new Map<string, typeof existing[number]>()
      const existingNamesByFolder = new Map<string, Set<string>>()
      for (const e of existing ?? []) {
        const folderKey = folderIdToName.get(e.folder_id) ?? ''
        const key = `${folderKey}::${e.name.toLowerCase()}`
        existingByKey.set(key, e)
        const set = existingNamesByFolder.get(folderKey) ?? new Set<string>()
        set.add(e.name.toLowerCase())
        existingNamesByFolder.set(folderKey, set)
      }

      // 3. Build ParsedRow-shaped input for dedupe
      const parsedRows: ParsedRow[] = input.rows.map((r, i) => ({
        rowNumber: i + 2, // approximate — client-side row numbers may differ; preserve in failedRows
        values: r,
        errors: [],
        isDuplicateInFile: false,
      }))

      const actions = applyDedupeStrategy({
        rows: parsedRows,
        existingKeys: new Set(existingByKey.keys()),
        existingNamesByFolder,
        globalStrategy: input.globalStrategy,
        perRowOverrides: new Map(
          Object.entries(input.perRowOverrides).map(([k, v]) => [Number(k), v])
        ),
      })

      // 4. Plan inserts + updates
      const toInsert: { row: ImportRow; finalName: string; folderKey: string }[] = []
      const toUpdate: { id: string; row: ImportRow; finalName: string; folderKey: string; prev: typeof existing[number] }[] = []
      let skippedCount = 0
      const failedRows: CommitChunkResult['failedRows'] = []

      for (const a of actions) {
        const rowIdx = a.rowNumber - 2
        const row = input.rows[rowIdx]
        if (!row) continue
        const folderKey = (row.folder_name ?? '').toLowerCase()
        if (a.action === 'skip') { skippedCount++; continue }
        if (a.action === 'insert') {
          toInsert.push({ row, finalName: a.finalName, folderKey })
        } else if (a.action === 'update') {
          const prev = existingByKey.get(`${folderKey}::${row.name.toLowerCase()}`)
          if (prev) toUpdate.push({ id: prev.id, row, finalName: a.finalName, folderKey, prev })
        }
      }

      let insertedCount = 0
      let updatedCount = 0
      const insertedItemIds: string[] = []
      const updatedItemIds: string[] = []
      const prevState: Record<string, { name: string; unit: string | null; unit_price: number; notes: string | null }> = {}

      if (!input.dryRun) {
        // 5. Insert chunk (already ≤50 rows enforced by client)
        if (toInsert.length > 0) {
          const payload = toInsert.map(({ row, finalName, folderKey }) => ({
            company_id: company.id,
            folder_id: folderMap.get(folderKey) ?? null,
            name: finalName,
            unit: row.unit || null,
            unit_price: row.unit_price,
            notes: row.notes || null,
          }))
          const { data: inserted, error: insErr } = await supabase
            .from('company_price_book')
            .insert(payload)
            .select('id')
          if (insErr) {
            failedRows.push(...toInsert.map((t) => ({
              rowNumber: 0, values: t.row, reason: `Insert failed: ${insErr.message}`,
            })))
          } else {
            insertedCount = inserted?.length ?? 0
            for (const r of inserted ?? []) insertedItemIds.push(r.id)
          }
        }

        // 6. Updates (per-row — can't batch with different values)
        for (const u of toUpdate) {
          prevState[u.id] = {
            name: u.prev.name, unit: u.prev.unit, unit_price: u.prev.unit_price, notes: u.prev.notes,
          }
          const { error: updErr } = await supabase
            .from('company_price_book')
            .update({
              unit: u.row.unit || null,
              unit_price: u.row.unit_price,
              notes: u.row.notes || null,
            })
            .eq('id', u.id)
          if (updErr) {
            failedRows.push({ rowNumber: 0, values: u.row, reason: `Update failed: ${updErr.message}` })
          } else {
            updatedCount++
            updatedItemIds.push(u.id)
          }
        }
      }

      // 7. Record/extend price_book_imports row
      const isFirstChunk = input.chunkIndex === 0
      let importId = input.importId
      const done = input.chunkIndex === input.totalChunks - 1

      if (!input.dryRun) {
        if (isFirstChunk) {
          const { data: imp, error: impErr } = await supabase
            .from('price_book_imports')
            .insert({
              company_id: company.id,
              actor_id: (await supabase.auth.getUser()).data.user?.id,
              inserted_item_ids: insertedItemIds,
              updated_item_ids: updatedItemIds,
              inserted_folder_ids: insertedFolderIds,
              prev_state: prevState,
              summary: {
                filename: input.filename,
                locale: input.locale,
                inserted: insertedCount,
                updated: updatedCount,
                skipped: skippedCount,
              },
            })
            .select('id')
            .single()
          if (impErr || !imp) return { error: 'Could not record import for undo.' }
          importId = imp.id
        } else if (importId) {
          // Merge into existing row
          const { data: existingImp } = await supabase
            .from('price_book_imports')
            .select('inserted_item_ids, updated_item_ids, inserted_folder_ids, prev_state, summary')
            .eq('id', importId)
            .single()
          if (existingImp) {
            const merged = {
              inserted_item_ids: [...(existingImp.inserted_item_ids ?? []), ...insertedItemIds],
              updated_item_ids: [...(existingImp.updated_item_ids ?? []), ...updatedItemIds],
              inserted_folder_ids: [...(existingImp.inserted_folder_ids ?? []), ...insertedFolderIds],
              prev_state: { ...(existingImp.prev_state as object ?? {}), ...prevState },
              summary: {
                ...(existingImp.summary as object ?? {}),
                inserted: ((existingImp.summary as { inserted?: number })?.inserted ?? 0) + insertedCount,
                updated: ((existingImp.summary as { updated?: number })?.updated ?? 0) + updatedCount,
                skipped: ((existingImp.summary as { skipped?: number })?.skipped ?? 0) + skippedCount,
              },
            }
            await supabase.from('price_book_imports').update(merged).eq('id', importId)
          }
        }
        if (done) revalidatePath('/price-book')
      }

      return {
        data: {
          importId: importId ?? 'dry-run',
          chunkIndex: input.chunkIndex,
          totalChunks: input.totalChunks,
          insertedCount,
          updatedCount,
          skippedCount,
          failedRows,
          done,
        },
      }
    }
    ```

    ```typescript
    export interface UndoResult { removedItems: number; removedFolders: number; revertedItems: number }

    export async function undoLastImport(
      importId: string
    ): Promise<{ data: UndoResult } | { error: string }> {
      const ctx = await getAuthContext()
      if ('error' in ctx) return { error: ctx.error }
      const { supabase, company } = ctx

      const { data: imp, error: impErr } = await supabase
        .from('price_book_imports')
        .select('id, company_id, created_at, inserted_item_ids, updated_item_ids, inserted_folder_ids, prev_state')
        .eq('id', importId)
        .eq('company_id', company.id)
        .single()
      if (impErr || !imp) return { error: 'Import not found.' }

      // 5-min eligibility gate (server-side)
      const createdMs = new Date(imp.created_at).getTime()
      if (Date.now() - createdMs > 5 * 60 * 1000) {
        return { error: 'Undo window expired (5 minutes).' }
      }

      let removedItems = 0
      let revertedItems = 0
      let removedFolders = 0

      // 1. Delete inserted items
      if (imp.inserted_item_ids && imp.inserted_item_ids.length > 0) {
        const { error: delErr, count } = await supabase
          .from('company_price_book')
          .delete({ count: 'exact' })
          .in('id', imp.inserted_item_ids)
          .eq('company_id', company.id)
        if (delErr) return { error: 'Could not remove inserted items.' }
        removedItems = count ?? 0
      }

      // 2. Revert updated items from prev_state
      const prev = (imp.prev_state ?? {}) as Record<string, { name: string; unit: string | null; unit_price: number; notes: string | null }>
      for (const id of imp.updated_item_ids ?? []) {
        const snap = prev[id]
        if (!snap) continue
        const { error: revErr } = await supabase
          .from('company_price_book')
          .update({ name: snap.name, unit: snap.unit, unit_price: snap.unit_price, notes: snap.notes })
          .eq('id', id)
          .eq('company_id', company.id)
        if (!revErr) revertedItems++
      }

      // 3. Delete folders created by this import (only if empty)
      if (imp.inserted_folder_ids && imp.inserted_folder_ids.length > 0) {
        for (const fId of imp.inserted_folder_ids) {
          const { count: itemCount } = await supabase
            .from('company_price_book')
            .select('id', { count: 'exact', head: true })
            .eq('company_id', company.id)
            .eq('folder_id', fId)
          if ((itemCount ?? 0) === 0) {
            const { error: delErr } = await supabase
              .from('price_book_folders')
              .delete()
              .eq('id', fId)
              .eq('company_id', company.id)
            if (!delErr) removedFolders++
          }
        }
      }

      // 4. Delete the import record itself
      await supabase.from('price_book_imports').delete().eq('id', importId)

      revalidatePath('/price-book')
      return { data: { removedItems, removedFolders, revertedItems } }
    }

    /**
     * Fetch the most recent import within the 5-min undo window (or null).
     * Used by /price-book page to show the Undo button after navigation.
     */
    export async function getRecentUndoableImport(): Promise<
      { data: { id: string; createdAt: string; summary: Record<string, unknown> } | null } | { error: string }
    > {
      const ctx = await getAuthContext()
      if ('error' in ctx) return { error: ctx.error }
      const { supabase, company } = ctx
      const fiveMinAgo = new Date(Date.now() - 5 * 60 * 1000).toISOString()
      const { data, error } = await supabase
        .from('price_book_imports')
        .select('id, created_at, summary')
        .eq('company_id', company.id)
        .gt('created_at', fiveMinAgo)
        .order('created_at', { ascending: false })
        .limit(1)
        .maybeSingle()
      if (error) return { error: 'Could not load recent imports.' }
      if (!data) return { data: null }
      return { data: { id: data.id, createdAt: data.created_at, summary: data.summary as Record<string, unknown> } }
    }
    ```

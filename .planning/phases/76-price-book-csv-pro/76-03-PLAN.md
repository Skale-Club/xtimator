    **`PriceBookImportWizard.tsx`** — orchestrator:

    ```typescript
    'use client'
    import { useEffect, useReducer, useRef, useState } from 'react'
    import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter } from '@/components/ui/dialog'
    import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle } from '@/components/ui/alert-dialog'
    import { Button } from '@/components/ui/button'
    import { Alert, AlertTitle, AlertDescription } from '@/components/ui/alert'
    import { History } from 'lucide-react'
    import { useTranslation } from 'react-i18next'
    import { wizardReducer, initialWizardState, serializeDraft, deserializeDraft, type WizardState } from '@/lib/csv/wizard-state'
    import { StepIndicator } from './StepIndicator'
    import { Step1Upload } from './Step1Upload'
    import { Step2Map } from './Step2Map'
    import { Step3Preview } from './Step3Preview'
    import { Step4Confirm } from './Step4Confirm'

    const DRAFT_KEY = 'xtimator:price-book-import:draft:v1'

    export interface PriceBookImportWizardProps {
      open: boolean
      onOpenChange: (open: boolean) => void
    }

    export function PriceBookImportWizard({ open, onOpenChange }: PriceBookImportWizardProps) {
      const { t } = useTranslation()
      const [state, dispatch] = useReducer(wizardReducer, initialWizardState)
      const [showCloseConfirm, setShowCloseConfirm] = useState(false)
      const [draftRestoredAt, setDraftRestoredAt] = useState<number | null>(null)
      const [draftAlertDismissed, setDraftAlertDismissed] = useState(false)
      const lastSavedRef = useRef<number>(0)

      // On open: try restore draft
      useEffect(() => {
        if (!open) return
        const json = sessionStorage.getItem(DRAFT_KEY)
        if (!json) return
        const restored = deserializeDraft(json)
        if (restored) {
          dispatch({ type: 'RESTORE_DRAFT', state: restored })
          setDraftRestoredAt(restored.savedAt)
          setDraftAlertDismissed(false)
        }
      }, [open])

      // Persist on state change (debounced — savedAt updates every action)
      useEffect(() => {
        if (state.savedAt === 0 || state.savedAt === lastSavedRef.current) return
        sessionStorage.setItem(DRAFT_KEY, serializeDraft(state))
        lastSavedRef.current = state.savedAt
      }, [state])

      const isMidProgress = state.step !== 'upload' || state.fileName !== null

      function handleClose() {
        if (!isMidProgress) {
          onOpenChange(false)
          return
        }
        setShowCloseConfirm(true)
      }

      function confirmSaveAndClose() {
        // draft already persisted on every dispatch — just close
        setShowCloseConfirm(false)
        onOpenChange(false)
      }

      function confirmDiscard() {
        sessionStorage.removeItem(DRAFT_KEY)
        dispatch({ type: 'RESET' })
        setShowCloseConfirm(false)
        onOpenChange(false)
      }

      function handleStartOver() {
        sessionStorage.removeItem(DRAFT_KEY)
        dispatch({ type: 'RESET' })
        setDraftRestoredAt(null)
      }

      function handleSuccessClose() {
        sessionStorage.removeItem(DRAFT_KEY)
        dispatch({ type: 'RESET' })
        onOpenChange(false)
      }

      return (
        <>
          <Dialog open={open} onOpenChange={(o) => { if (!o) handleClose(); else onOpenChange(true) }}>
            <DialogContent className="sm:max-w-3xl max-w-[calc(100%-2rem)] max-h-[85vh] flex flex-col">
              <DialogHeader>
                <DialogTitle>{t(`pricebook.import.${state.step}.title`)}</DialogTitle>
                <DialogDescription>{t(`pricebook.import.${state.step}.description`)}</DialogDescription>
              </DialogHeader>

              <StepIndicator
                currentStep={state.step}
                onStepClick={(step) => {
                  // only allow backtracking to completed steps
                  const order = ['upload', 'map', 'preview', 'confirm'] as const
                  if (order.indexOf(step) < order.indexOf(state.step)) {
                    dispatch({ type: 'RESTORE_DRAFT', state: { ...state, step } })
                  }
                }}
              />

              {draftRestoredAt && !draftAlertDismissed && (
                <Alert>
                  <History className="h-4 w-4" />
                  <AlertTitle>{t('pricebook.import.draft.restored_title')}</AlertTitle>
                  <AlertDescription className="flex items-center justify-between gap-2">
                    <span>{t('pricebook.import.draft.restored_body')}</span>
                    <Button variant="ghost" size="sm" onClick={handleStartOver}>
                      {t('pricebook.import.draft.start_over')}
                    </Button>
                  </AlertDescription>
                </Alert>
              )}

              <div className="flex-1 min-h-0 overflow-y-auto">
                {state.step === 'upload' && <Step1Upload state={state} dispatch={dispatch} />}
                {state.step === 'map' && <Step2Map state={state} dispatch={dispatch} />}
                {state.step === 'preview' && <Step3Preview state={state} dispatch={dispatch} />}
                {state.step === 'confirm' && <Step4Confirm state={state} dispatch={dispatch} onDone={handleSuccessClose} />}
              </div>

              <DialogFooter>
                {state.step !== 'upload' && (
                  <Button variant="ghost" onClick={() => dispatch({ type: 'BACK' })}>
                    {t('pricebook.import.footer.back')}
                  </Button>
                )}
                <div className="flex-1" />
                <Button variant="ghost" onClick={handleClose}>
                  {t('pricebook.import.footer.cancel')}
                </Button>
                {/* Primary CTA is rendered per-step inside step components for label flexibility */}
              </DialogFooter>
            </DialogContent>
          </Dialog>

          <AlertDialog open={showCloseConfirm} onOpenChange={setShowCloseConfirm}>
            <AlertDialogContent>
              <AlertDialogHeader>
                <AlertDialogTitle>{t('pricebook.import.close_confirm.title')}</AlertDialogTitle>
                <AlertDialogDescription>{t('pricebook.import.close_confirm.body')}</AlertDialogDescription>
              </AlertDialogHeader>
              <AlertDialogFooter>
                <Button variant="ghost" onClick={confirmDiscard}>
                  {t('pricebook.import.close_confirm.discard')}
                </Button>
                <AlertDialogAction onClick={confirmSaveAndClose}>
                  {t('pricebook.import.close_confirm.save')}
                </AlertDialogAction>
              </AlertDialogFooter>
            </AlertDialogContent>
          </AlertDialog>
        </>
      )
    }
    ```

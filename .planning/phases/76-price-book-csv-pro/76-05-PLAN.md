    ```typescript
    'use client'
    import { useState } from 'react'
    import { Button } from '@/components/ui/button'
    import { Card } from '@/components/ui/card'
    import { Progress } from '@/components/ui/progress'
    import { Alert, AlertDescription } from '@/components/ui/alert'
    import { Check, AlertTriangle, Download } from 'lucide-react'
    import { toast } from 'sonner'
    import { useTranslation } from 'react-i18next'
    import { useRouter } from 'next/navigation'
    import { commitImportChunk } from '@/lib/actions/price-book'
    import { buildErrorCsv } from '@/lib/csv/error-csv'
    import type { WizardState, WizardAction } from '@/lib/csv/wizard-state'
    import type { ImportRow } from '@/lib/csv/price-book-import'

    const CHUNK_SIZE = 50

    interface Props {
      state: WizardState
      dispatch: React.Dispatch<WizardAction>
      onDone: () => void
    }

    type SubView =
      | { kind: 'pre' }
      | { kind: 'importing'; done: number; total: number }
      | { kind: 'success'; inserted: number; updated: number; skipped: number; importId: string; failedRows: { values: ImportRow; reason: string }[] }
      | { kind: 'failure'; done: number; message: string; failedRows: { values: ImportRow; reason: string }[] }

    export function Step4Confirm({ state, dispatch: _dispatch, onDone }: Props) {
      const { t } = useTranslation()
      const router = useRouter()
      const [view, setView] = useState<SubView>({ kind: 'pre' })
      const [cancelRef] = useState({ aborted: false })

      // Apply cell edits to rows
      const editedRows: ImportRow[] = state.rows
        .filter((r) => r.errors.length === 0 && !r.isDuplicateInFile)
        .map((r) => {
          const edits = state.cellEdits[r.rowNumber] ?? {}
          return { ...r.values, ...edits } as ImportRow
        })

      const total = editedRows.length
      const chunkCount = Math.max(1, Math.ceil(total / CHUNK_SIZE))

      async function startImport() {
        setView({ kind: 'importing', done: 0, total })
        cancelRef.aborted = false

        let importId: string | undefined
        let insertedTotal = 0, updatedTotal = 0, skippedTotal = 0
        const failedRowsAll: { values: ImportRow; reason: string }[] = []

        for (let i = 0; i < chunkCount; i++) {
          if (cancelRef.aborted) {
            setView({ kind: 'failure', done: i * CHUNK_SIZE, message: 'Canceled', failedRows: failedRowsAll })
            return
          }
          const slice = editedRows.slice(i * CHUNK_SIZE, (i + 1) * CHUNK_SIZE)
          const res = await commitImportChunk({
            rows: slice,
            globalStrategy: state.globalDedupeStrategy,
            perRowOverrides: state.perRowDedupeOverrides,
            chunkIndex: i,
            totalChunks: chunkCount,
            importId,
            filename: state.fileName ?? 'import.csv',
            locale: state.locale,
          })
          if ('error' in res) {
            setView({ kind: 'failure', done: i * CHUNK_SIZE, message: res.error, failedRows: failedRowsAll })
            return
          }
          importId = res.data.importId
          insertedTotal += res.data.insertedCount
          updatedTotal += res.data.updatedCount
          skippedTotal += res.data.skippedCount
          failedRowsAll.push(...res.data.failedRows.map((f) => ({ values: f.values, reason: f.reason })))
          setView({ kind: 'importing', done: Math.min(total, (i + 1) * CHUNK_SIZE), total })
        }

        setView({
          kind: 'success',
          inserted: insertedTotal,
          updated: updatedTotal,
          skipped: skippedTotal,
          importId: importId!,
          failedRows: failedRowsAll,
        })
      }

      function downloadErrorCsv(failedRows: { values: ImportRow; reason: string }[]) {
        const csv = buildErrorCsv(failedRows, state.headers)
        const blob = new Blob([csv], { type: 'text/csv' })
        const url = URL.createObjectURL(blob)
        const a = document.createElement('a')
        a.href = url
        a.download = `import-errors-${Date.now()}.csv`
        a.click()
        URL.revokeObjectURL(url)
      }

      if (view.kind === 'pre') {
        return (
          <div className="max-w-md mx-auto space-y-6">
            <Card className="p-6 text-center">
              <p className="text-xs uppercase text-muted-foreground tracking-wide">{t('pricebook.import.confirm.stat_label')}</p>
              <p className="text-4xl font-semibold mt-2">{total}</p>
            </Card>
            <Card className="p-4 space-y-2">
              <div className="flex justify-between"><span>{t('pricebook.import.confirm.breakdown_new')}</span><span>{total}</span></div>
              <div className="flex justify-between"><span>{t('pricebook.import.confirm.breakdown_skip')}</span><span>{state.rows.filter(r => r.isDuplicateInFile).length}</span></div>
              <div className="flex justify-between"><span>{t('pricebook.import.confirm.breakdown_errors')}</span><span>{state.rows.filter(r => r.errors.length > 0).length}</span></div>
            </Card>
            <p className="text-xs text-muted-foreground text-center">
              {t('pricebook.import.confirm.source_line', { filename: state.fileName, locale: state.locale })}
            </p>
            <div className="flex justify-end">
              <Button variant="primary" onClick={startImport} disabled={total === 0} data-testid="commit-import-btn">
                {t('pricebook.import.confirm.primary_cta', { n: total })}
              </Button>
            </div>
          </div>
        )
      }

      if (view.kind === 'importing') {
        const pct = view.total === 0 ? 0 : (view.done / view.total) * 100
        return (
          <div className="max-w-md mx-auto space-y-4">
            <Card className="p-6 space-y-3">
              <h3 className="font-semibold">{t('pricebook.import.confirm.progress_headline')}</h3>
              <Progress value={pct} className="h-2" />
              <p className="text-sm">{t('pricebook.import.confirm.progress_sub', { done: view.done, total: view.total, eta: '' })}</p>
              <div className="flex justify-end">
                <Button variant="outline" size="sm" onClick={() => { cancelRef.aborted = true }}>
                  {t('pricebook.import.confirm.cancel_button')}
                </Button>
              </div>
            </Card>
          </div>
        )
      }

      if (view.kind === 'success') {
        return (
          <div className="max-w-md mx-auto text-center space-y-4">
            <div className="w-12 h-12 mx-auto rounded-full bg-gradient-success flex items-center justify-center">
              <Check className="w-6 h-6 text-white" />
            </div>
            <h2 className="text-xl font-semibold">{t('pricebook.import.confirm.success_h2', { n: view.inserted + view.updated })}</h2>
            <p className="text-muted-foreground">{t('pricebook.import.confirm.success_body', { newCount: view.inserted, updateCount: view.updated })}</p>
            <div className="flex flex-col gap-2">
              <Button variant="primary" onClick={() => { toast.success(t('pricebook.import.confirm.undo_toast', { n: view.inserted + view.updated })); router.push('/price-book'); onDone() }}>
                {t('pricebook.import.confirm.success_primary')}
              </Button>
              {view.failedRows.length > 0 && (
                <Button variant="outline" onClick={() => downloadErrorCsv(view.failedRows)}>
                  <Download className="w-4 h-4 mr-2" />
                  {t('pricebook.import.confirm.success_error_report', { n: view.failedRows.length })}
                </Button>
              )}
            </div>
          </div>
        )
      }

      // failure
      return (
        <div className="max-w-md mx-auto text-center space-y-4">
          <div className="w-12 h-12 mx-auto rounded-full bg-gradient-danger flex items-center justify-center">
            <AlertTriangle className="w-6 h-6 text-white" />
          </div>
          <h2 className="text-xl font-semibold">{t('pricebook.import.confirm.failure_h2')}</h2>
          <Alert variant="destructive">
            <AlertDescription>{t('pricebook.import.confirm.failure_body', { done: view.done, serverMessage: view.message })}</AlertDescription>
          </Alert>
          <div className="flex flex-col gap-2">
            <Button variant="primary" onClick={startImport}>{t('pricebook.import.confirm.failure_primary')}</Button>
            {view.failedRows.length > 0 && (
              <Button variant="outline" onClick={() => downloadErrorCsv(view.failedRows)}>
                <Download className="w-4 h-4 mr-2" />
                {t('pricebook.import.confirm.success_error_report', { n: view.failedRows.length })}
              </Button>
            )}
            <Button variant="ghost" onClick={onDone}>{t('pricebook.import.footer.cancel')}</Button>
          </div>
        </div>
      )
    }
    ```

    ```typescript
    'use client'
    import { useEffect, useState } from 'react'
    import { Undo2 } from 'lucide-react'
    import { toast } from 'sonner'
    import { useRouter } from 'next/navigation'
    import { useTranslation } from 'react-i18next'
    import { Alert, AlertDescription } from '@/components/ui/alert'
    import { Button } from '@/components/ui/button'
    import { getRecentUndoableImport, undoLastImport } from '@/lib/actions/price-book'

    export function UndoImportBanner() {
      const { t } = useTranslation()
      const router = useRouter()
      const [imp, setImp] = useState<{ id: string; createdAt: string; summary: Record<string, unknown> } | null>(null)

      useEffect(() => {
        getRecentUndoableImport().then((res) => {
          if ('data' in res) setImp(res.data)
        })
      }, [])

      // Tick to hide once 5 min elapses without a refresh
      useEffect(() => {
        if (!imp) return
        const elapsed = Date.now() - new Date(imp.createdAt).getTime()
        const remaining = 5 * 60 * 1000 - elapsed
        if (remaining <= 0) { setImp(null); return }
        const tId = setTimeout(() => setImp(null), remaining)
        return () => clearTimeout(tId)
      }, [imp])

      if (!imp) return null

      const summary = imp.summary as { inserted?: number; updated?: number }
      const n = (summary.inserted ?? 0) + (summary.updated ?? 0)

      async function handleUndo() {
        const res = await undoLastImport(imp!.id)
        if ('error' in res) {
          toast.error(res.error)
          return
        }
        toast.success(`Removed ${res.data.removedItems} items, reverted ${res.data.revertedItems}.`)
        setImp(null)
        router.refresh()
      }

      return (
        <Alert className="flex items-center justify-between">
          <AlertDescription>{t('pricebook.import.confirm.undo_toast', { n })}</AlertDescription>
          <Button size="sm" variant="outline" onClick={handleUndo}>
            <Undo2 className="w-4 h-4 mr-2" /> Undo
          </Button>
        </Alert>
      )
    }
    ```

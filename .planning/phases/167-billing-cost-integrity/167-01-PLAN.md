```ts
// lib/billing/whisper-cost.ts — the derivation seam (pure, unit-tested)
// OPUS BLOCKER FIX: browsers record with NO audioBitsPerSecond set → real bitrate is the browser
// default, commonly 64-128 kbps (NOT 32). A low assumed bitrate would FALSELY UP-CLAMP honest
// recordings (overcharge + false entitlement rejections). Therefore:
//   - assume the HIGH end: bitrateBps default 128_000 → byteEstimate is a conservative LOWER BOUND
//     on true minutes (any real bitrate <=128k makes declared >= byteEstimate for honest audio)
//   - clamp ONE-SIDED for billing: use byteEstimate ONLY when declared < byteEstimate (under-declaration
//     — the exploit); NEVER up-clamp beyond declared, and do NOT down-clamp an over-declaration
//     (declaring more than true only hurts the declarer; platform never loses)
export function deriveAudioMinutes(input: {
  declaredSeconds: number | null
  blobBytes: number
  bitrateBps?: number       // default 128_000 (conservative HIGH assumption — lower bound estimate)
}): { minutes: number; source: 'declared' | 'byte_clamp' }
// byteEstimateMinutes = blobBytes * 8 / bitrateBps / 60
// declared null/<=0 OR declared < byteEstimate → byteEstimate (source 'byte_clamp')
// else → declared (source 'declared')
// The 1s-on-10-min exploit: declared 0.017 min vs byteEstimate ≈ 2.5-10 min → byte_clamp catches it
// under ANY bitrate assumption; an honest 5-min take at 96kbps: byteEstimate ≈ 3.75 min < declared 5 → declared. Safe.
```

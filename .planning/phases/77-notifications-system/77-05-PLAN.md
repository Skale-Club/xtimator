**`lib/notifications/queries.ts`** — server-only:

```typescript
import 'server-only'
import { requireServiceClient } from '@/lib/supabase/service'
import type { EventCategory, EventType } from './event-types'
import { EVENT_CATEGORIES } from './event-types'

export interface ListNotificationsArgs {
  companyId: string
  userId: string
  category?: EventCategory | null
  unreadOnly?: boolean
  cursor?: string | null
  limit?: number
}

export async function listNotificationsPage(args: ListNotificationsArgs) {
  const svc = requireServiceClient()
  const limit = Math.min(args.limit ?? 50, 100)

  let q = svc
    .from('notifications')
    .select('*')
    .eq('company_id', args.companyId)
    .or(`user_id.is.null,user_id.eq.${args.userId}`)
    .order('pinned', { ascending: false })
    .order('created_at', { ascending: false })
    .limit(limit + 1) // fetch one extra to derive nextCursor

  if (args.cursor) q = q.lt('created_at', args.cursor)
  if (args.unreadOnly) q = q.is('read_at', null)
  if (args.category) {
    const types = (Object.keys(EVENT_CATEGORIES) as EventType[])
      .filter(t => EVENT_CATEGORIES[t] === args.category)
    q = q.in('event_type', types)
  }

  const { data, error } = await q
  if (error) throw error
  const items = (data ?? []).slice(0, limit)
  const nextCursor = (data?.length ?? 0) > limit ? items[items.length - 1].created_at : null
  return { items, nextCursor }
}
```

**`app/api/notifications/page/route.ts`** — GET with query params:
```typescript
import { NextRequest, NextResponse } from 'next/server'
import { getAuthClaims } from '@/lib/queries/auth'
import { listNotificationsPage } from '@/lib/notifications/queries'

export async function GET(req: NextRequest) {
  const claims = await getAuthClaims()
  if (!claims) return NextResponse.json({ error: 'unauth' }, { status: 401 })
  const sp = req.nextUrl.searchParams
  const data = await listNotificationsPage({
    companyId: claims.company_id,
    userId: claims.sub,
    category: (sp.get('category') as any) || null,
    unreadOnly: sp.get('unread') === '1',
    cursor: sp.get('cursor'),
    limit: 50,
  })
  return NextResponse.json(data)
}
```

import { redirect } from 'next/navigation'
import { getAuthClaims } from '@/lib/queries/auth'
import { listNotificationsPage } from '@/lib/notifications/queries'
import { NotificationsClient } from './notifications-client'

export const metadata = { title: 'Notifications' }

export default async function NotificationsPage() {
  const claims = await getAuthClaims()
  if (!claims) redirect('/login')
  const initial = await listNotificationsPage({
    companyId: claims.company_id,
    userId: claims.sub,
    limit: 50,
  })
  return (
    <div className="container max-w-4xl py-8">
      <h1 className="text-2xl font-semibold mb-6">Notifications</h1>
      <NotificationsClient initialItems={initial.items} initialNextCursor={initial.nextCursor} />
    </div>
  )
}

Create 3 API routes. All use `createClient()` from `@/lib/supabase/server` (RLS enforces scope automatically — NOT the service client).

**GET /api/notifications/list** — query params `?cursor=<iso>&limit=20`. Selects from `notifications` where (company_id, user_id) match auth (RLS handles); orders `pinned DESC, created_at DESC`; returns `{items, nextCursor}`.

**PATCH /api/notifications/[id]/read** — uses **service client** to update `read_at = now()` for the row (RLS blocks client-side updates). Verify the caller owns it: `WHERE id=$id AND company_id = jwt.company_id AND (user_id IS NULL OR user_id = jwt.sub)`. Returns 204.

**POST /api/notifications/mark-all-read** — service client; updates all `read_at IS NULL` rows scoped to caller's company + user. Returns `{updated: number}`.

Pattern for the read endpoint:

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { getAuthClaims } from '@/lib/queries/auth'
import { requireServiceClient } from '@/lib/supabase/service'

export async function PATCH(_req: NextRequest, { params }: { params: Promise<{ id: string }> }) {
  const claims = await getAuthClaims()
  if (!claims) return NextResponse.json({ error: 'unauth' }, { status: 401 })
  const { id } = await params
  const svc = requireServiceClient()
  const { error } = await svc
    .from('notifications')
    .update({ read_at: new Date().toISOString() })
    .eq('id', id)
    .eq('company_id', claims.company_id)
    .or(`user_id.is.null,user_id.eq.${claims.sub}`)
  if (error) return NextResponse.json({ error: error.message }, { status: 500 })
  return new NextResponse(null, { status: 204 })
}
```

```typescript
'use client'
import { useEffect, useState, useCallback, useRef } from 'react'
import { createBrowserClient } from '@/lib/supabase/client' // existing browser-client helper
import type { EventCategory } from '@/lib/notifications/event-types'
import { EVENT_CATEGORIES } from '@/lib/notifications/event-types'

export interface NotificationListItem {
  id: string
  event_type: string
  category: EventCategory
  title: string
  body: string
  link_url: string | null
  created_at: string
  read_at: string | null
  pinned: boolean
}

interface UseNotificationsArgs { companyId: string; userId: string }

export function useNotifications({ companyId, userId }: UseNotificationsArgs) {
  const [items, setItems] = useState<NotificationListItem[]>([])
  const [loading, setLoading] = useState(true)
  const debounceRef = useRef<NodeJS.Timeout | null>(null)
  const pendingRef = useRef<NotificationListItem[]>([])

  const flush = useCallback(() => {
    if (pendingRef.current.length === 0) return
    setItems(prev => [...pendingRef.current, ...prev])
    pendingRef.current = []
  }, [])

  const refresh = useCallback(async () => {
    setLoading(true)
    const res = await fetch('/api/notifications/list?limit=20')
    const data = await res.json()
    setItems(data.items.map((r: any) => ({ ...r, category: EVENT_CATEGORIES[r.event_type as keyof typeof EVENT_CATEGORIES] ?? 'system' })))
    setLoading(false)
  }, [])

  useEffect(() => { refresh() }, [refresh])

  useEffect(() => {
    const supabase = createBrowserClient()
    const channel = supabase
      .channel(`notifications:${companyId}:${userId}`)
      .on(
        'postgres_changes',
        {
          event: 'INSERT',
          schema: 'public',
          table: 'notifications',
          filter: `company_id=eq.${companyId}`,
        },
        (payload: any) => {
          const row = payload.new
          // Filter: must be user-scoped or company-wide
          if (row.user_id && row.user_id !== userId) return
          const item: NotificationListItem = {
            ...row,
            category: EVENT_CATEGORIES[row.event_type as keyof typeof EVENT_CATEGORIES] ?? 'system',
          }
          pendingRef.current.push(item)
          if (debounceRef.current) clearTimeout(debounceRef.current)
          debounceRef.current = setTimeout(flush, 300)
        }
      )
      .subscribe()
    return () => {
      supabase.removeChannel(channel)
      if (debounceRef.current) clearTimeout(debounceRef.current)
    }
  }, [companyId, userId, flush])

  const markRead = useCallback(async (id: string) => {
    setItems(prev => prev.map(i => i.id === id ? { ...i, read_at: new Date().toISOString() } : i))
    await fetch(`/api/notifications/${id}/read`, { method: 'PATCH' })
  }, [])

  const markAllRead = useCallback(async () => {
    const now = new Date().toISOString()
    setItems(prev => prev.map(i => i.read_at ? i : { ...i, read_at: now }))
    await fetch('/api/notifications/mark-all-read', { method: 'POST' })
  }, [])

  const unreadCount = items.filter(i => !i.read_at).length
  return { items, unreadCount, loading, markRead, markAllRead, refresh }
}
```

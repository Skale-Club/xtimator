  <action>
Two triggers in one function (Inngest supports array of triggers):
1. **Event-driven** (`notification/email.queued` from 77-02) — for immediate one-off mode
2. **Cron** (`*/15 * * * *`) — for digest mode (groups when >3 events/15min/category)

Logic:
- If triggered by event: check if there are ≥3 unsent items for the same user+category in the last 15 minutes (via service-role query on notifications joined to a `notification_email_sent` tracking — OR simpler: track a `metadata.email_sent_at` field). If yes, do NOTHING (the cron will batch). If no, send immediately and mark sent.
- If triggered by cron: query all notifications with no email-sent marker, created in last 15 min, grouped by (user_id, category). For each group with count ≥1, look up user email + branding, call `sendNotificationDigestEmail`, then mark items as sent.

Track "email sent" via `metadata.email_sent_at` JSONB patch on the notifications row (no schema change needed).

```typescript
import { inngest } from '@/lib/inngest/client'
import { requireServiceClient } from '@/lib/supabase/service'
import { sendNotificationDigestEmail } from '@/lib/email/notification-emails'
import { getBranding } from '@/lib/platform-config'
import { EVENT_CATEGORIES, type EventType } from '@/lib/notifications/event-types'

export const notificationEmailDigest = inngest.createFunction(
  { id: 'notification-email-digest', idempotency: 'event.data.notificationId' },
  [
    { event: 'notification/email.queued' },
    { cron: '*/15 * * * *' },
  ],
  async ({ event, step }) => {
    const svc = requireServiceClient()
    const fifteenMinAgo = new Date(Date.now() - 15 * 60 * 1000).toISOString()

    // Cron path: batch unsent in last 15min
    const { data: rows } = await svc
      .from('notifications')
      .select('id, user_id, company_id, event_type, title, body, link_url, created_at, metadata')
      .gte('created_at', fifteenMinAgo)
      .not('user_id', 'is', null)
      .filter('metadata->>email_sent_at', 'is', null)

    if (!rows || rows.length === 0) return { sent: 0 }

    // Group by (user_id, category)
    const groups = new Map<string, typeof rows>()
    for (const r of rows) {
      const cat = EVENT_CATEGORIES[r.event_type as EventType] ?? 'system'
      const key = `${r.user_id}|${cat}`
      const list = groups.get(key) ?? []
      list.push(r)
      groups.set(key, list)
    }

    let sent = 0
    for (const [key, items] of groups) {
      const [userId] = key.split('|')
      // Look up user email + branding for the company
      const { data: user } = await svc.auth.admin.getUserById(userId!)
      if (!user?.user?.email) continue
      const branding = await getBranding() // tenant-aware via existing helper
      await step.run(`send-digest-${key}`, async () => {
        await sendNotificationDigestEmail({
          toEmail: user.user!.email!,
          toName: user.user!.user_metadata?.full_name ?? null,
          branding: { logoUrl: branding.logo_url ?? null, brandColor: branding.brand_color ?? null, businessName: branding.business_name ?? 'Xtimator' },
          items: items.map(i => ({
            category: EVENT_CATEGORIES[i.event_type as EventType] ?? 'system',
            title: i.title,
            body: i.body,
            linkUrl: i.link_url ?? undefined,
            createdAt: i.created_at,
          })),
          groupedByCategory: items.length > 3,
        })
      })
      // Mark sent
      const sentAt = new Date().toISOString()
      await svc
        .from('notifications')
        .update({ metadata: { email_sent_at: sentAt } })
        .in('id', items.map(i => i.id))
      sent += items.length
    }

    return { sent, groups: groups.size }
  }
)
```

NOTE: The metadata update above naively overwrites — in practice merge with existing metadata. Use Postgres `jsonb_set` via raw RPC if needed, or fetch + spread + write back.

```typescript
import { inngest } from '@/lib/inngest/client'
import { requireServiceClient } from '@/lib/supabase/service'

export const notificationCleanup = inngest.createFunction(
  { id: 'notification-cleanup' },
  { cron: '0 3 * * *' }, // daily 03:00 UTC
  async ({ step }) => {
    const svc = requireServiceClient()
    const sixtyDaysAgo = new Date(Date.now() - 60 * 24 * 60 * 60 * 1000).toISOString()
    const nowIso = new Date().toISOString()

    const { data, error } = await svc
      .from('notifications')
      .delete({ count: 'exact' })
      .lt('created_at', sixtyDaysAgo)
      .eq('pinned', false)
      .or(`expires_at.is.null,expires_at.lt.${nowIso}`)
      .select('id')

    if (error) {
      console.warn('[notification-cleanup] failed:', error.message)
      return { deleted: 0, error: error.message }
    }
    return { deleted: data?.length ?? 0 }
  }
)
```

Create `lib/notifications/copy.ts` with a switch over all 17 EventTypes; each returns short, action-oriented strings. Example pattern:

```typescript
export function buildNotificationCopy(eventType: EventType, ctx: CopyContext): NotificationCopy {
  switch (eventType) {
    case 'estimate.viewed':
      return {
        title: 'Estimate viewed',
        body: `${ctx.clientName ?? 'A client'} opened estimate ${ctx.estimateNumber ?? ''}`.trim(),
      }
    case 'estimate.accepted':
      return { title: 'Estimate accepted', body: `${ctx.clientName ?? 'Client'} accepted ${ctx.estimateNumber ?? 'your estimate'}` }
    case 'estimate.declined':
      return { title: 'Estimate declined', body: `${ctx.clientName ?? 'Client'} declined ${ctx.estimateNumber ?? 'your estimate'}` }
    case 'estimate.expired':
      return { title: 'Estimate expired', body: `Estimate ${ctx.estimateNumber ?? ''} reached its expiry without a response` }
    case 'payment.received':
      return { title: 'Payment received', body: `${ctx.amountUSD ?? ''} received for ${ctx.projectName ?? 'a project'}` }
    case 'payment.refunded':
      return { title: 'Payment refunded', body: `${ctx.amountUSD ?? ''} refunded for ${ctx.projectName ?? 'a project'}` }
    case 'trial.expiring_3d':
      return { title: 'Trial ends soon', body: `Your trial ends in ${ctx.daysRemaining ?? 3} days. Upgrade to keep Pro features.` }
    case 'trial.expired':
      return { title: 'Trial expired', body: 'Your trial has ended. You have been moved to the free plan.' }
    case 'trial.converted':
      return { title: 'Welcome to Pro', body: 'Your subscription is active. Thanks for upgrading!' }
    case 'quota.80pct':
      return { title: 'Approaching monthly limit', body: `You've used ${ctx.quotaPercent ?? 80}% of your monthly AI quota.` }
    case 'quota.exhausted':
      return { title: 'Monthly quota exhausted', body: 'You have used all included AI credits this month. Upgrade or wait until reset.' }
    case 'whatsapp.inbound':
      return { title: 'New WhatsApp message', body: `Message from ${ctx.whatsappFrom ?? 'a contact'}` }
    case 'ai_job.failed':
      return { title: 'AI job failed', body: `${ctx.jobType ?? 'Job'} did not complete: ${ctx.errorMessage ?? 'unknown error'}` }
    case 'ai_job.completed':
      return { title: 'AI job complete', body: `${ctx.jobType ?? 'Job'} finished successfully.` }
    case 'admin.tier_changed':
      return { title: 'Plan updated by admin', body: `Your plan was changed from ${ctx.tierFrom ?? '?'} to ${ctx.tierTo ?? '?'}.` }
    case 'admin.bonus_credits_granted':
      return { title: 'Bonus credits granted', body: `An admin granted you ${ctx.credits ?? 0} bonus credits.` }
    case 'system.maintenance':
      return { title: ctx.maintenanceTitle ?? 'Scheduled maintenance', body: ctx.maintenanceBody ?? 'Brief downtime expected.' }
  }
}
```

// after the existing business write...
const copy = buildNotificationCopy('estimate.viewed', { estimateNumber, clientName })
await notify({
  companyId,
  userId: ownerUserId,
  eventType: 'estimate.viewed',
  title: copy.title,
  body: copy.body,
  linkUrl: `/projects/${projectId}/estimates/${estimateId}`,

})

    ```tsx
    import { cn } from "@/lib/utils"
    import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
    import { Button } from "@/components/ui/button"
    import { Badge } from "@/components/ui/badge"

    export type Tier = 'free' | 'pro' | 'business'

    interface TierCardProps {
      tier: Tier
      name: string
      price: string
      period: string
      features: string[]
      ctaLabel: string
      onSelect: () => void
      current?: boolean
      popular?: boolean
    }

    export function TierCard({ tier, name, price, period, features, ctaLabel, onSelect, current, popular }: TierCardProps) {
      const variant = tier === 'pro' ? 'stat' : 'glass'
      return (
        <Card variant={variant} className={cn(
          "relative p-8 flex flex-col gap-6",
          tier === 'business' && "before:content-[''] before:absolute before:inset-x-0 before:top-0 before:h-[3px] before:bg-[image:var(--gradient-premium)] before:rounded-t-[var(--radius-lg)] overflow-hidden",
        )}>
          {popular && (
            <Badge variant="brand" className="absolute -top-2 left-1/2 -translate-x-1/2">Most popular</Badge>
          )}
          <CardHeader className="p-0">
            <CardTitle className="text-2xl">{name}</CardTitle>
          </CardHeader>
          <div className="flex items-baseline gap-1">
            <span className="font-mono text-4xl font-semibold">{price}</span>
            <span className="text-muted-foreground text-sm">/ {period}</span>
          </div>
          <ul className="space-y-2 text-sm">
            {features.map(f => <li key={f} className="flex items-start gap-2"><span className="text-[hsl(var(--primary))]">✓</span>{f}</li>)}
          </ul>
          <Button
            variant={tier === 'free' ? 'outline' : tier === 'pro' ? 'primary' : 'premium'}
            onClick={onSelect}
            disabled={current}
            className="w-full"
          >
            {current ? 'Current plan' : ctaLabel}
          </Button>
        </Card>
      )
    }
    ```

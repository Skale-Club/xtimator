    test.describe('@visual capture', () => {
      test.skip(!hasAuth || !seedProjectId, 'auth fixture or SEED_PROJECT_ID missing')
      const screens = ['capture', 'describe', 'photos-input']
      for (const screen of screens) {
        for (const vp of viewports) {
          for (const lang of (['en', 'pt'] as const)) {
            test(`${screen} — ${vp.name} ${lang}`, async ({ page, context }) => {
              await setLang(context, lang)
              await page.setViewportSize({ width: vp.width, height: vp.height })
              await page.goto(`/projects/${seedProjectId}/${screen}`, { waitUntil: 'networkidle' })

              // NOTE: the gradient ring uses framer-motion; freezeAnimations stops it so snapshot is stable
              await expect(page).toHaveScreenshot(`capture-${screen}-${vp.name}-${lang}.png`, { fullPage: true })

**BUG-04 — lib/actions/auth.ts: updatePassword must check company existence before redirecting**

Current `updatePassword` function (lines 87-103):
```ts
export async function updatePassword(formData: FormData) {
  const supabase = await createClient()
  const password = formData.get('password') as string

  const { error } = await supabase.auth.updateUser({ password })

  if (error) {
    logAuthEvent({ event: 'password_update', success: false, error: error.message })
    if (error.message.includes('expired') || error.message.includes('invalid')) {
      return { error: 'This reset link has expired. Request a new one.' }
    }
    return { error: 'Something went wrong. Please try again.' }
  }

  logAuthEvent({ event: 'password_update', success: true })
  redirect('/dashboard')
}
```

Replace the success path (after logAuthEvent) with:
```ts
  logAuthEvent({ event: 'password_update', success: true })

  // Check company record to determine redirect destination (same pattern as signIn / callback)
  const { data } = await supabase.auth.getClaims()
  const claims = data?.claims ?? null
  if (claims) {
    const { data: company } = await supabase
      .from('companies')
      .select('id')
      .eq('user_id', claims.sub)
      .single()
    redirect(company ? '/dashboard' : '/onboarding')
  }

  // Fallback: no claims available immediately after updateUser (edge case)
  redirect('/login')
```

**BUG-06 — components/auth/google-oauth-button.tsx: reset isLoading on failure/cancel**

Current `handleGoogleSignIn` function:
```ts
  async function handleGoogleSignIn() {
    setIsLoading(true)
    const supabase = createClient()
    await supabase.auth.signInWithOAuth({
      provider: 'google',
      options: {
        redirectTo: `${window.location.origin}/callback`,
      },
    })
    // Loading stays true while redirect happens
  }
```

Replace with:
```ts
  async function handleGoogleSignIn() {
    setIsLoading(true)
    try {
      const supabase = createClient()
      const { error } = await supabase.auth.signInWithOAuth({
        provider: 'google',
        options: {
          redirectTo: `${window.location.origin}/callback`,
        },
      })
      // On success, browser navigates away — this code never executes.
      // On failure (error returned or no redirect URL), reset loading state.
      if (error) {
        setIsLoading(false)
      }
    } catch {
      // Network error, popup cancelled, or unexpected failure — reset loading state.
      setIsLoading(false)
    }
  }
```

Rationale: `signInWithOAuth` on success triggers a browser redirect — the code after the `await` never runs for the happy path. But if OAuth fails (Supabase returns an error), the user is NOT redirected, so `isLoading` stays `true` forever and the button remains permanently disabled. The try/catch + error check resets the button in all failure cases without affecting the successful redirect flow.

    Create `lib/storage/s3-provider.ts`:
    ```typescript
    /**
     * STORAGE-05: S3-compatible StorageProvider implementation.
     *
     * Works against any S3-compatible backend: AWS S3, MinIO (local dev),
     * Hetzner Object Storage (target — see docs/STORAGE-MIGRATION.md), Cloudflare R2, etc.
     *
     * Activated via env var `STORAGE_PROVIDER=s3` (see lib/storage/index.ts).
     * Default remains Supabase — this provider ships as a working skeleton, not
     * the runtime default, until the migration trigger fires (800 MB Supabase usage).
     */
    import {
      S3Client,
      PutObjectCommand,
      GetObjectCommand,
      DeleteObjectCommand,
      ListObjectsV2Command,
    } from '@aws-sdk/client-s3'
    import { getSignedUrl } from '@aws-sdk/s3-request-presigner'
    import type { StorageProvider, StorageBody, UploadOptions, ListedObject } from './index'

    export interface S3ProviderConfig {
      endpoint: string                  // e.g. https://fsn1.your-objectstorage.com (Hetzner) or http://localhost:9000 (MinIO)
      region: string                    // e.g. fsn1 (Hetzner) or us-east-1 (MinIO ignores)
      accessKeyId: string
      secretAccessKey: string
      forcePathStyle?: boolean          // true for MinIO + most non-AWS S3
      publicUrlBase?: string            // e.g. https://fsn1.your-objectstorage.com — used for getPublicUrl()
    }

    export function createS3StorageProvider(config: S3ProviderConfig): StorageProvider {
      const client = new S3Client({
        endpoint: config.endpoint,
        region: config.region,
        credentials: {
          accessKeyId: config.accessKeyId,
          secretAccessKey: config.secretAccessKey,
        },
        forcePathStyle: config.forcePathStyle ?? true,
      })
      const publicBase = config.publicUrlBase ?? config.endpoint

      return {
        async upload(bucket, path, body, opts) {
          const Body = await normalizeBody(body)
          await client.send(new PutObjectCommand({
            Bucket: bucket,
            Key: path,
            Body,
            ContentType: opts?.contentType,
            // Note: S3 has no native upsert toggle — PutObject is upsert by default.
            // The opts.upsert=false in Supabase calls is best-effort emulated by checking existence
            // first if strict semantics are required; for now, document this behavioral diff.
          }))
          return { path }
        },

        async download(bucket, path) {
          const res = await client.send(new GetObjectCommand({ Bucket: bucket, Key: path }))
          if (!res.Body) throw new Error(`S3 download empty (${bucket}/${path})`)
          // Body is a stream in Node — convert to Blob
          const bytes = await res.Body.transformToByteArray()
          return new Blob([bytes])
        },

        async getSignedUrl(bucket, path, expiresInSeconds) {
          if (!expiresInSeconds || expiresInSeconds <= 0) {
            throw new Error('getSignedUrl: expiresInSeconds must be a positive integer (STORAGE-04)')
          }
          return getSignedUrl(client, new GetObjectCommand({ Bucket: bucket, Key: path }), {
            expiresIn: expiresInSeconds,
          })
        },

        getPublicUrl(bucket, path) {
          // Path-style URL (works for MinIO + Hetzner): {endpoint}/{bucket}/{key}
          return `${publicBase.replace(/\/$/, '')}/${bucket}/${path}`
        },

        async delete(bucket, path) {
          await client.send(new DeleteObjectCommand({ Bucket: bucket, Key: path }))
        },

        async list(bucket, prefix) {
          const res = await client.send(new ListObjectsV2Command({
            Bucket: bucket,
            Prefix: prefix,
          }))
          return (res.Contents ?? []).map((obj) => ({
            name: obj.Key ?? '',
            size: obj.Size,
            updatedAt: obj.LastModified?.toISOString(),
          }))
        },
      }
    }

    async function normalizeBody(body: StorageBody): Promise<Buffer | Uint8Array> {
      if (body instanceof Buffer) return body
      if (body instanceof Uint8Array) return body
      if (body instanceof ArrayBuffer) return new Uint8Array(body)
      // Blob or File
      const ab = await (body as Blob).arrayBuffer()
      return Buffer.from(ab)
    }
    ```

    ```typescript
    /**
     * Server-side default storage provider.
     *
     * Reads `process.env.STORAGE_PROVIDER`:
     *   - 'supabase' (default, omitted, or any other value): wraps the service-role Supabase client
     *   - 's3': uses createS3StorageProvider() with S3_* env vars
     *
     * Use this in server-only contexts where the Supabase client isn't already in hand
     * (e.g. health check, smoke script, future Inngest workers).
     *
     * Browser code MUST continue using `createStorage(supabaseBrowserClient)` —
     * the env-var path is server-only because S3 credentials must never reach the browser.
     */
    export function getServerStorage(): StorageProvider {
      const provider = process.env.STORAGE_PROVIDER ?? 'supabase'
      if (provider === 's3') {
        // Lazy import so the AWS SDK isn't loaded unless actually needed
        const { createS3StorageProvider } = require('./s3-provider') as typeof import('./s3-provider')
        return createS3StorageProvider({
          endpoint: requireEnv('S3_ENDPOINT'),
          region: requireEnv('S3_REGION'),
          accessKeyId: requireEnv('S3_ACCESS_KEY_ID'),
          secretAccessKey: requireEnv('S3_SECRET_ACCESS_KEY'),
          forcePathStyle: process.env.S3_FORCE_PATH_STYLE !== 'false',
          publicUrlBase: process.env.S3_PUBLIC_URL_BASE,
        })
      }
      // Default: Supabase via service client
      const { requireServiceClient } = require('@/lib/supabase/service') as typeof import('@/lib/supabase/service')
      return createStorage(requireServiceClient())
    }

    function requireEnv(name: string): string {
      const v = process.env[name]
      if (!v) throw new Error(`getServerStorage: missing required env var ${name} (STORAGE_PROVIDER=s3)`)
      return v
    }

    ```typescript
    /**
     * Storage smoke test — STORAGE-07.
     *
     * Exercises the four core operations against the configured provider:
     *   1. upload   — write a small text payload
     *   2. getSignedUrl — produce a 60-second URL
     *   3. download — fetch via the signed URL AND directly via storage.download
     *   4. delete   — clean up
     *
     * Usage:
     *   # Against Supabase (default — uses your dev Supabase service role)
     *   npx tsx scripts/storage-smoke.ts
     *
     *   # Against local MinIO:
     *   docker run -d -p 9000:9000 -p 9001:9001 \
     *     -e MINIO_ROOT_USER=minioadmin -e MINIO_ROOT_PASSWORD=minioadmin \
     *     --name minio minio/minio server /data --console-address ":9001"
     *   # Then in MinIO console (http://localhost:9001) create a bucket called 'smoketest'.
     *   STORAGE_PROVIDER=s3 \
     *     S3_ENDPOINT=http://localhost:9000 \
     *     S3_REGION=us-east-1 \
     *     S3_ACCESS_KEY_ID=minioadmin \
     *     S3_SECRET_ACCESS_KEY=minioadmin \
     *     npx tsx scripts/storage-smoke.ts smoketest
     *
     * After running, verify Supabase remains the default by un-setting STORAGE_PROVIDER
     * (or restarting your dev server with the original .env.local).
     */
    import { getServerStorage } from '@/lib/storage'

    const BUCKET = process.argv[2] ?? 'pdfs'
    const KEY = `smoke/${Date.now()}-roundtrip.txt`
    const PAYLOAD = `storage smoke test at ${new Date().toISOString()}`

    async function main() {
      const storage = getServerStorage()
      const provider = process.env.STORAGE_PROVIDER ?? 'supabase'
      console.log(`[smoke] provider=${provider} bucket=${BUCKET} key=${KEY}`)

      // 1. upload
      await storage.upload(BUCKET, KEY, Buffer.from(PAYLOAD), { contentType: 'text/plain' })
      console.log('[smoke] upload OK')

      // 2. signed URL
      const signed = await storage.getSignedUrl(BUCKET, KEY, 60)
      console.log(`[smoke] signed URL OK (${signed.slice(0, 60)}...)`)

      // 3a. download via storage API
      const blob = await storage.download(BUCKET, KEY)
      const text = await blob.text()
      if (text !== PAYLOAD) throw new Error(`[smoke] FAIL download mismatch — got: ${text}`)
      console.log('[smoke] download OK (content roundtrip verified)')

      // 3b. download via signed URL (HTTP fetch)
      const resp = await fetch(signed)
      if (!resp.ok) throw new Error(`[smoke] FAIL signed-URL fetch ${resp.status}`)
      const httpText = await resp.text()
      if (httpText !== PAYLOAD) throw new Error(`[smoke] FAIL signed-URL content mismatch`)
      console.log('[smoke] signed-URL fetch OK')

      // 4. delete
      await storage.delete(BUCKET, KEY)
      console.log('[smoke] delete OK')

      console.log('[smoke] ALL OPS PASSED')
    }

    main().catch((err) => { console.error('[smoke] FAIL', err); process.exit(1) })

    # Storage Migration: Supabase → Hetzner Object Storage

    Status: Documented, NOT executed. Trigger threshold: **Supabase Storage usage exceeds 800 MB** (the soft point where the Free tier becomes uncomfortable; well below the 1 GB hard cap).

    ## Why this is a 1-line change
    Phase 66 introduced `lib/storage/` — every call site routes through `StorageProvider`.
    Switching providers means flipping `STORAGE_PROVIDER=s3` and supplying `S3_*` env vars.
    No application code changes. No call site changes.

    ## When to trigger
    - Hard trigger: Supabase Storage usage hits 800 MB (check via Supabase dashboard → Storage → Usage)
    - Soft trigger: Egress costs become noticeable on the Supabase bill
    - Strategic trigger: Moving to Hetzner Cloud VPS (Phase 68 artifacts) — co-locate storage with compute

    ## Pre-migration checklist
    - [ ] Storage usage measured and recorded
    - [ ] All five buckets enumerated: `audio`, `photos`, `pdfs`, `logos`, `platform-brand`
    - [ ] Maintenance window scheduled (writes paused during sync — typical 30-60 min depending on size)
    - [ ] Hetzner account in good standing
    - [ ] Daily Supabase backup confirmed taken (Phase 61 baseline)

    ## Step 1 — Provision Hetzner Object Storage
    1. Log in to Hetzner Cloud Console → Object Storage
    2. Create a new project (or use existing)
    3. Choose region — recommend `fsn1` (Falkenstein) or `nbg1` (Nuremberg) for EU latency
    4. Create five buckets matching current Supabase bucket names:
       - `audio`, `photos`, `pdfs`, `logos`, `platform-brand`
    5. Generate S3 credentials: Project Settings → S3 Credentials → Generate
       - Save Access Key ID + Secret Access Key in your password manager
       - Hetzner endpoint shape: `https://<region>.your-objectstorage.com`

    ## Step 2 — Sync data
    Use `aws-cli` v2 (install: `brew install awscli` / `apt install awscli`).

    Configure two profiles:
    ```bash
    aws configure --profile supabase
    # AWS Access Key ID:     <supabase-storage-s3-access-key>     # placeholder shape: s3_access_key_<your-key>
    # AWS Secret Access Key: <supabase-storage-s3-secret>         # placeholder shape: s3_secret_<your-key>
    # Default region:        us-east-1
    # Default output:        json

    aws configure --profile hetzner
    # AWS Access Key ID:     <hetzner-objectstorage-access-key>   # placeholder shape: s3_access_key_<your-key>
    # AWS Secret Access Key: <hetzner-objectstorage-secret>       # placeholder shape: s3_secret_<your-key>
    # Default region:        fsn1
    # Default output:        json
    ```

    Run sync per bucket (this is the load-bearing command for STORAGE-06):
    ```bash
    SUPABASE_S3_ENDPOINT="https://<your-supabase-project-ref>.supabase.co/storage/v1/s3"
    HETZNER_S3_ENDPOINT="https://fsn1.your-objectstorage.com"

    for BUCKET in audio photos pdfs logos platform-brand; do
      echo "=== Syncing $BUCKET ==="
      aws s3 sync \
        s3://$BUCKET \
        s3://$BUCKET \
        --source-region us-east-1 \
        --region fsn1 \
        --profile supabase \
        --endpoint-url "$SUPABASE_S3_ENDPOINT" \
        --no-progress \
        --only-show-errors
      # Then re-sync source → destination via the destination profile
      aws --profile hetzner --endpoint-url "$HETZNER_S3_ENDPOINT" \
        s3 sync s3://$BUCKET s3://$BUCKET
    done
    ```

    Verify object counts match per bucket:
    ```bash
    aws --profile supabase --endpoint-url "$SUPABASE_S3_ENDPOINT" s3 ls s3://photos --recursive | wc -l
    aws --profile hetzner  --endpoint-url "$HETZNER_S3_ENDPOINT"  s3 ls s3://photos --recursive | wc -l
    ```

    ## Step 3 — Swap the application provider (the 1-line change)
    Add to production env (Vercel dashboard → Settings → Environment Variables, or the .env on the Hetzner VPS):
    ```bash
    STORAGE_PROVIDER=s3
    S3_ENDPOINT=https://fsn1.your-objectstorage.com
    S3_REGION=fsn1
    S3_ACCESS_KEY_ID=s3_access_key_<your-hetzner-key>
    S3_SECRET_ACCESS_KEY=s3_secret_<your-hetzner-key>
    S3_FORCE_PATH_STYLE=true
    S3_PUBLIC_URL_BASE=https://fsn1.your-objectstorage.com
    ```

    Redeploy (Vercel: trigger redeploy; Hetzner VPS: `docker compose up -d --force-recreate`).

    Validate: `curl https://your-domain/api/health` should return `storage: 'ok'` (Phase 68 health check uses the new provider).

    ## Step 4 — Smoke test in production
    Run a controlled UAT-style smoke:
    1. Generate one estimate with audio + 1 photo — confirm both upload successfully
    2. Send the estimate as PDF via WhatsApp — confirm signed URL returns the PDF
    3. Open the share link — confirm the estimate page renders with the company logo (logos bucket)
    4. Inbound WhatsApp message with a photo — confirm it lands in the photos bucket and is processed

    ## Step 5 — Decommission Supabase Storage
    After 7 days of incident-free operation:
    1. Take a final backup snapshot of the Supabase buckets (`aws s3 sync` to local `./supabase-storage-backup/`)
    2. Delete Supabase Storage buckets via dashboard
    3. Remove Supabase Storage RLS policies from migrations (cosmetic — they target deleted buckets)

    ## Rollback procedure
    If anything misbehaves in the first 24 hours:
    1. Set `STORAGE_PROVIDER=supabase` (or remove the env var) and redeploy
    2. App returns to reading from Supabase Storage immediately (data was never deleted from Supabase during steps 1-4)
    3. Investigate the failure offline; re-attempt migration after fix

    ## Cost reference (informational)
    | Item                  | Supabase (Free / Pro) | Hetzner Object Storage |
    |-----------------------|----------------------|------------------------|
    | Storage / GB / month  | included up to 1 GB  | EUR 0.0095             |
    | Egress / GB           | 5 GB free / 0.09 EUR | EUR 0.01               |
    | API calls             | unmetered            | unmetered              |

    ## What to NEVER do
    - Do NOT delete Supabase Storage buckets in the same maintenance window as the cutover. Wait the full 7 days.
    - Do NOT commit real S3 credentials anywhere — see CLAUDE.md Secret Handling. Only `.env.production` on the deploy target.
    - Do NOT swap providers without first running `scripts/storage-smoke.ts` against the destination (caught by Phase 66 STORAGE-07 in dev — re-run against Hetzner before flipping production).
    ```

    export default async function ChatPage({ params }: { params: Promise<{ id?: string[] }> }) {

          activeId={conversationId ?? null}
          initialMessages={initialMessages}
        />
      )
    }

    export function ChatThread({ conversationId, initialMessages }: { conversationId: string | null; initialMessages: UIMessage[] }) {
      const router = useRouter()
      const [activeId, setActiveId] = useState(conversationId)
      const [text, setText] = useState('')
      const { messages, sendMessage, status } = useChat({
        id: activeId ?? undefined,
        messages: initialMessages,
        transport: new DefaultChatTransport({ api: '/api/chat', body: { conversationId: activeId } }),
      })
      const endRef = useRef<HTMLDivElement>(null)

          if (conv) { id = conv.id; setActiveId(id); router.replace(`/chat/${id}`) }
